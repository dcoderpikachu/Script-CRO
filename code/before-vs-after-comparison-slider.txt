<script>
  function initImageComparisons(container = document) {
    const comparisons = container.querySelectorAll('ed-comparison-before-after:not([data-initialized])');

    comparisons.forEach((wrapper) => {
      const button = wrapper.querySelector('.ed_comparison__button');
      const after = wrapper.querySelector('.ed_comparison__after');
      const before = wrapper.querySelector('.ed_comparison__before');
      const texts = wrapper.querySelectorAll('.ed_comparison__box');

      if (!button || !after || !before) return;

      wrapper.dataset.initialized = 'true'; // Prevent double-init

      let isDragging = false;

      const updateSlider = (xPercent) => {
        const rect = wrapper.getBoundingClientRect();
        const buttonRect = button.getBoundingClientRect();
        const buttonWidthPercent = (buttonRect.width / rect.width) * 100;

        // Limit slider to stay within visible bounds of the button
        const min = buttonWidthPercent / 2;
        const max = 100 - min;
        xPercent = Math.max(min, Math.min(xPercent, max));

        button.style.left = `${xPercent}%`;
        after.style.clipPath = `inset(0 ${100 - xPercent}% 0 0)`;
      };

      const fadeText = (show) => {
        texts.forEach((el) => {
          el.style.transition = 'opacity 0.3s ease';
          el.style.opacity = show ? '1' : '0';
        });
      };

      const onDragStart = (e) => {
        isDragging = true;
        fadeText(false);
        e.preventDefault();
      };

      const onDragMove = (e) => {
        if (!isDragging) return;
        const rect = wrapper.getBoundingClientRect();
        const clientX = e.touches ? e.touches[0].clientX : e.clientX;
        const percent = ((clientX - rect.left) / rect.width) * 100;
        updateSlider(percent);
      };

      const onDragEnd = () => {
        isDragging = false;
        fadeText(true);
      };

      button.addEventListener('mousedown', onDragStart);
      button.addEventListener('touchstart', onDragStart, { passive: false });
      window.addEventListener('mousemove', onDragMove);
      window.addEventListener('touchmove', onDragMove, { passive: false });
      window.addEventListener('mouseup', onDragEnd);
      window.addEventListener('touchend', onDragEnd);

      updateSlider(50);
      fadeText(true);
    });
  }

  document.addEventListener('DOMContentLoaded', () => {
    initImageComparisons();
  });

  if (typeof Shopify !== 'undefined' && Shopify.designMode) {
    document.addEventListener('shopify:section:load', (event) => {
      initImageComparisons(event.target);
    });
  }
</script>
