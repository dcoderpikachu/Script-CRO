  <script>
    const miniCart = document.getElementById("mini-cart");

    function checkCheckoutButtonVisibility() {
      {% if section.settings.enable_sticky_button %}
      const checkoutButton = document.querySelector("#checkout-button-wrapper");
      let checkoutButtonClone = document.querySelector("#checkout-button-clone");

      if (checkoutButton) {
        const rect = checkoutButton.getBoundingClientRect();
        const isVisible = rect.top < window.innerHeight && rect.bottom > 0;

        if (!checkoutButtonClone) {
          checkoutButtonClone = checkoutButton.cloneNode(true);
          checkoutButtonClone.id = "checkout-button-clone";
          checkoutButtonClone.classList.add("sticky-checkout-btn");
          if (miniCart) miniCart.appendChild(checkoutButtonClone);
        }

        if (checkoutButtonClone) {
          checkoutButtonClone.style.display = isVisible ? "none" : "block";
        }
      }
      {% endif %}

      {% if section.settings.enable_sticky_paymenticons %}
      const paymentIcons = document.querySelector("#payment-icons-and-store-benefits");
      let paymentIconsClone = document.querySelector("#ed-payment-bar-cart-clone");

      if (paymentIcons) {
        const rect2 = paymentIcons.getBoundingClientRect();
        const isVisible2 = rect2.top < window.innerHeight && rect2.bottom > 0;

        if (!paymentIconsClone) {
          paymentIconsClone = paymentIcons.cloneNode(true);
          paymentIconsClone.id = "ed-payment-bar-cart-clone";
          paymentIconsClone.classList.add("sticky");
          document.querySelector("#mini-cart")?.appendChild(paymentIconsClone);
        }

  if (checkoutButtonClone && paymentIconsClone) {
    const height = paymentIconsClone.offsetHeight;
    checkoutButtonClone.style.setProperty('--payment-icon-height', `${height}px`);
}

        if (paymentIconsClone) {
          paymentIconsClone.style.display = isVisible2 ? "none" : "block";
        }
      }
      {% endif %}
    }

    if (miniCart) {
      checkCheckoutButtonVisibility();
      miniCart.addEventListener("scroll", checkCheckoutButtonVisibility);
    }
  </script>
{% endif %}
