Create a new snippet: pdp_product_badges.liquid
Paste this code:

{% if product.metafields.standard.pdp_product_badge_01 != blank
  or product.metafields.standard.pdp_product_badge_02 != blank
  or product.metafields.standard.pdp_product_badge_03 != blank
%}
  <div class="pdp_title_with_badges {{ block.settings.pdp_badge_position }}">
    <style>
      {% if block.settings.custom_font_family %}
          {% assign font_url = block.settings.font_family | font_url %}
          @font-face {
              font-family: '{{ block.settings.font_family.family }}';
              src: url({{ font_url }}) format('woff2');
              font-weight: {{ block.settings.font_family.weight }};
              font-style: normal;
          }
          .pdp_title_with_badges {
            font-family: '{{ block.settings.font_family.family }}';
            font-weight: {{ block.settings.font_family.weight }};
          }
      {% endif %}
      .pdp_title_with_badges {
        margin-top: {{ block.settings.pdp_badge_space_top -}}px;
        margin-bottom: {{ block.settings.pdp_badge_space_bottom }}px;
        display: flex;
        flex-direction: column;
        gap: {{ block.settings.badge_distance_with_title }}px;
      }

      .pdp_title_with_badges .product-meta__title.heading {
        margin: 0px;
      }

      .pdp_badges_list {
        display: flex;
        flex-wrap: wrap;
        gap: 8px;
      }

      .pdp_badges_list .pdp_badge {
        font-size: {{ block.settings.badge_text_size -}}px;
        line-height: 1.1;
        font-weight: {{ block.settings.badge_font_weight }};
        padding: {{  block.settings.badge_tb_padding }}px {{ block.settings.badge_lr_padding }}px;
        border-radius: {{ block.settings.badge_corner_radius }}px;
        border: 1px solid transparent;
        display: flex;
        align-items: center;
        gap: 4px;
        {% if block.settings.badge_text_uppercase %}
          text-transform: uppercase;
        {% endif %}
      }
      .pdp_badges_list .pdp_badge.badge_1 {
        color: {{ block.settings.color_badge_1 }};
        border-color: {{ block.settings.border_badge_1 }};
        background: {{ block.settings.background_badge_1 }};
      }
      .pdp_badges_list .pdp_badge.badge_2 {
        color: {{ block.settings.color_badge_2 }};
        border-color: {{ block.settings.border_badge_2 }};
        background: {{ block.settings.background_badge_2 }};
      }
      .pdp_badges_list .pdp_badge.badge_3 {
        color: {{ block.settings.color_badge_3 }};
        border-color: {{ block.settings.border_badge_3 }};
        background: {{ block.settings.background_badge_3 }};
      }
      .pdp_badges_list .pdp_badge img {
          width: 14px;
          height: auto;
          display: flex;
      }

      .pdp_title_with_badges.top-left .pdp_badges_list {
          justify-content: flex-start;
      }
      .pdp_title_with_badges.top-right .pdp_badges_list {
          justify-content: flex-end;
      }
      .pdp_title_with_badges.below-title .pdp_badges_list {
          order: 2;
      }

      @media (max-width: 767px) {
        .pdp_badges_list .pdp_badge {
            font-size: {{ block.settings.badge_text_size_mobile -}}px;
        }
      }
    </style>
    {% unless featured %}
      <div class="pdp_badges_list">
        {%- for i in (1..3) -%}
          {%- assign image_key = 'icon_badge_' | append: i -%}
          {%- assign text_key = 'pdp_product_badge_0' | append: i -%}
          {%- assign badge_image = block.settings[image_key] -%}
          {%- assign text = product.metafields.standard[text_key] -%}

          <div class="pdp_badge badge_{{ forloop.index }}">
            {% if block.settings.show_badge_icon %}
              {%- if badge_image != blank -%}
                {{-
                  badge_image
                  | image_url: width: badge_image.width
                  | image_tag: sizes: '100vw', widths: '20', alt: badge_image.alt
                -}}
              {%- endif -%}
            {% endif %}
            <span>{{ text }}</span>
          </div>
        {%- endfor -%}
      </div>
      <h1 class="product-meta__title heading h3">{{ product.title }}</h1>
    {% else %}
      <h2 class="product-meta__title heading h3">
        <a href="{{ product.url }}">{{ product.title }}</a>
      </h2>
    {% endunless %}
  </div>
{% endif %}
