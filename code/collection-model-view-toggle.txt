<script>
    function initModelView() {
        const toggleInput = document.getElementById("ed-model-view-input");
        const productList = document.querySelector(".product-list.product-facet__product-list");
        const openButton = document.getElementById("ed-model-view-button-open");
        const closeButton = document.getElementById("ed-model-view-button-close");

        function updateView(isModelView) {
            productList.classList.toggle("ed-ModelView-active", isModelView);
            openButton.classList.toggle("active", isModelView);
            closeButton.classList.toggle("active", !isModelView);
            if (toggleInput) {
            toggleInput.checked = isModelView;
            }
        }

        if (toggleInput && productList) {
            toggleInput.addEventListener("change", function () {
            updateView(this.checked);
            });
        }

        if (openButton && closeButton) {
            openButton.addEventListener("click", function () {
            updateView(true);
            });

            closeButton.addEventListener("click", function () {
            updateView(false);
            });
        }
    }

    document.addEventListener("DOMContentLoaded", function () {
        initModelView();
    });

    if (typeof Shopify !== "undefined" && Shopify.designMode) {
      document.addEventListener("shopify:section:load", function () {
        initModelView();
      });
    }
</script>
