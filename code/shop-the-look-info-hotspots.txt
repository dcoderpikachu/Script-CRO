<script>
  /* SKB Universal Modal JS */
  (function () {
    window.initSKBModal = function (scope = document) {
      const toggleButtons = scope.querySelectorAll('[is="skb-popup-toggle-button"]');

      toggleButtons.forEach((button) => {
        const targetId = button.getAttribute('aria-controls');
        const modal = scope.querySelector(`#${targetId}`);
        if (!modal || !modal.tagName.toLowerCase().includes('skb-modal-content')) return;

        const overlay = modal.querySelector('.skb_popup_modal__overlay');
        const closeBtn = modal.querySelector('[data-action="close"]');
        const modalContent = modal.querySelector('.skb_popup_modal__content');

        function openModal() {
          modal.setAttribute('open', '');
          button.setAttribute('aria-expanded', 'true');
          if (window.innerWidth <= 999) {
            document.documentElement.classList.add('lock-all');
          }
        }

        function closeModal() {
          modal.removeAttribute('open');
          button.setAttribute('aria-expanded', 'false');
          if (window.innerWidth <= 999) {
            document.documentElement.classList.remove('lock-all');
          }
        }

        // Only toggle the modal open/close for screen sizes <= 1000px
        if (window.innerWidth >= 1000) {
          button.addEventListener('click', function () {
            // Toggle the modal's open/close state when the button is clicked
            if (modal.hasAttribute('open')) {
              closeModal();
            } else {
              openModal();
            }
          });
        } else {
          // For larger screens, just open the modal when the button is clicked
          button.addEventListener('click', openModal);
        }

        closeBtn?.addEventListener('click', closeModal);
        overlay?.addEventListener('click', closeModal);

        document.addEventListener('keydown', function (e) {
          if (e.key === 'Escape' && modal.hasAttribute('open')) {
            closeModal();
          }
        });

        // Add the logic for closing the modal on click outside for screens larger than 1000px
        if (window.innerWidth > 1000) {
          // Exclude the click on the open modal button
          document.addEventListener('click', function (e) {
            // If the click is not inside the modal content or the modal itself, and the modal is open, close it
            if (
              modal.hasAttribute('open') &&
              !modalContent.contains(e.target) &&
              !modal.contains(e.target) &&
              !button.contains(e.target)
            ) {
              closeModal();
            }
          });
        }
      });
    };

    document.addEventListener('DOMContentLoaded', () => {
      initSKBModal();
    });

    if (typeof Shopify !== 'undefined' && Shopify.designMode) {
      document.addEventListener('shopify:section:load', (e) => {
        initSKBModal(e.target);
      });
    }
  })();
</script>
