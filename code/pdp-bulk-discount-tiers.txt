  // Universal Multi-Tabs
  (function () {
    function initSKBUniversalMultiTabs(scope = document) {
      const containers = scope.querySelectorAll('.skb_universal_tabs_wrap');

      containers.forEach((container) => {
        const tabs = container.querySelectorAll('[data-skb-tab-button]');
        const contents = container.querySelectorAll('[data-skb-tab-content]');
        const indicator = container.querySelector('.skb_tab_nav__position'); // Optional

        // Move indicator under active tab
        function moveIndicator(tab) {
          if (!indicator) return; // Skip if indicator is not found
          const tabRect = tab.getBoundingClientRect();
          const parentRect = tab.parentElement.getBoundingClientRect();
          const leftOffset = tabRect.left - parentRect.left;
          indicator.style.width = `${tabRect.width}px`;
          indicator.style.transform = `translateX(${leftOffset}px)`;
        }

        // Set initial state
        const firstActiveTab = container.querySelector('.skb_tab.skb_active') || tabs[0];
        if (firstActiveTab) {
          firstActiveTab.classList.add('skb_active');
          moveIndicator(firstActiveTab);
          const firstTarget = firstActiveTab.getAttribute('data-skb-tab-button');
          const firstContent = container.querySelector(`[data-skb-tab-content="${firstTarget}"]`);
          if (firstContent) {
            firstContent.classList.add('skb_show');
            firstContent.style.position = 'relative';
          }
        }

        // Add click events
        tabs.forEach((tab) => {
          tab.addEventListener('click', () => {
            const target = tab.getAttribute('data-skb-tab-button');

            tabs.forEach((t) => t.classList.remove('skb_active'));
            tab.classList.add('skb_active');

            moveIndicator(tab);

            contents.forEach((content) => {
              content.classList.remove('skb_show');
              content.style.position = 'absolute';
            });

            const next = container.querySelector(`[data-skb-tab-content="${target}"]`);
            if (next) {
              next.classList.add('skb_show');
              next.style.position = 'relative';
            }
          });
        });

        // Recalculate on window resize
        if (indicator) {
          window.addEventListener('resize', () => {
            const activeTab = container.querySelector('.skb_tab.skb_active');
            if (activeTab) moveIndicator(activeTab);
          });
        }
      });
    }

    // DOM ready
    document.addEventListener('DOMContentLoaded', () => {
      initSKBUniversalMultiTabs();
    });

    // Shopify section reload support
    if (typeof Shopify !== 'undefined' && Shopify.designMode) {
      document.addEventListener('shopify:section:load', (e) => {
        initSKBUniversalMultiTabs(e.target);
      });
    }
  })();
</script>
