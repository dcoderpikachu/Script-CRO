<script>
  // IIFE Function to create custom counter elements
  (function () {
    // Create a unique scope to avoid conflicts
    const counters = document.querySelectorAll('skb-counter-text');

    if (!('IntersectionObserver' in window)) return;

    const observer = new IntersectionObserver(
      (entries, obs) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            const counterElement = entry.target;
            const targetValue = parseInt(counterElement.getAttribute('data-counter'), 10);
            const duration = 3000; // 2 seconds total animation time
            const stepTime = Math.max(Math.floor(duration / targetValue), 20);
            let current = 0;

            const interval = setInterval(() => {
              current++;
              counterElement.textContent = current;
              if (current >= targetValue) {
                clearInterval(interval);
              }
            }, stepTime);

            obs.unobserve(counterElement);
          }
        });
      },
      {
        threshold: 0.5,
      }
    );

    counters.forEach((counter) => {
      observer.observe(counter);
    });
  })();
</script>
