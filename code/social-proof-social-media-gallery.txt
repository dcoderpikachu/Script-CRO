  (function () {
    function initSkbModalGalleryInstance(gallery) {
      // Avoid initializing the same gallery more than once
      if (gallery.dataset.skbInitialized === 'true') return;
      gallery.dataset.skbInitialized = 'true';

      const htmlEl = document.documentElement;
      const items = [...gallery.querySelectorAll('.skb_modal_gallery__item')];
      const nextBtn = gallery.querySelector('.skb_modal_nav.next');
      const prevBtn = gallery.querySelector('.skb_modal_nav.prev');
      let currentIndex = -1;

      function closeAllModals() {
        items.forEach((item) => {
          const modal = item.querySelector('.skb_modal_gallery__item_details');
          if (modal) modal.removeAttribute('open');
        });
        if (!document.querySelector('.skb_modal_gallery__item_details[open]')) {
          htmlEl.classList.remove('lock-all');
        }
      }

      function switchModal(index) {
        closeAllModals();
        const item = items[index];
        const modal = item.querySelector('.skb_modal_gallery__item_details');
        if (modal) {
          modal.setAttribute('open', '');
          if (!htmlEl.classList.contains('lock-all')) {
            htmlEl.classList.add('lock-all');
          }
          currentIndex = index;
        }
      }

      items.forEach((item, index) => {
        const openBtn = item.querySelector('.skb_modal_gallery__item_button');
        const modal = item.querySelector('.skb_modal_gallery__item_details');
        const closeBtn = item.querySelector('.skb_modal_gallery__close');
        const contentWrap = item.querySelector('.skb_mg_item_details_wrap');

        if (openBtn && modal) {
          openBtn.addEventListener('click', () => {
            closeAllModals();
            modal.setAttribute('open', '');
            if (!htmlEl.classList.contains('lock-all')) {
              htmlEl.classList.add('lock-all');
            }
            currentIndex = index;
          });
        }

        if (closeBtn && modal) {
          closeBtn.addEventListener('click', () => {
            modal.removeAttribute('open');
            if (!document.querySelector('.skb_modal_gallery__item_details[open]')) {
              htmlEl.classList.remove('lock-all');
            }
          });
        }

        if (modal && contentWrap) {
          modal.addEventListener('click', (e) => {
            if (!contentWrap.contains(e.target)) {
              modal.removeAttribute('open');
              if (!document.querySelector('.skb_modal_gallery__item_details[open]')) {
                htmlEl.classList.remove('lock-all');
              }
            }
          });
        }
      });

      if (nextBtn) {
        nextBtn.addEventListener('click', () => {
          if (currentIndex === -1) return;
          currentIndex = (currentIndex + 1) % items.length;
          switchModal(currentIndex);
        });
      }

      if (prevBtn) {
        prevBtn.addEventListener('click', () => {
          if (currentIndex === -1) return;
          currentIndex = (currentIndex - 1 + items.length) % items.length;
          switchModal(currentIndex);
        });
      }

      // Keyboard navigation (specific to this gallery)
      document.addEventListener('keydown', function handleKeydown(e) {
        const isOpen = items.some((item) =>
          item.querySelector('.skb_modal_gallery__item_details')?.hasAttribute('open')
        );
        if (!isOpen) return;

        if (e.key === 'ArrowRight' && nextBtn) {
          currentIndex = (currentIndex + 1) % items.length;
          switchModal(currentIndex);
        } else if (e.key === 'ArrowLeft' && prevBtn) {
          currentIndex = (currentIndex - 1 + items.length) % items.length;
          switchModal(currentIndex);
        } else if (e.key === 'Escape') {
          closeAllModals();
        }
      });
    }

    function initAllGalleriesInDocument(context = document) {
      const galleries = context.querySelectorAll('.skb_modal_gallery');
      galleries.forEach(initSkbModalGalleryInstance);
    }

    // Run on initial load
    document.addEventListener('DOMContentLoaded', () => {
      initAllGalleriesInDocument();
    });

    // Run on Shopify section load (dynamic)
    if (typeof Shopify !== 'undefined' && Shopify.designMode) {
      document.addEventListener('shopify:section:load', function (event) {
        const container = event.target;
        if (container) initAllGalleriesInDocument(container);
      });
    }
  })();

  (function () {
    function loadResource(type, url) {
      return new Promise((resolve) => {
        if (document.querySelector(`${type === 'css' ? 'link' : 'script'}[href="${url}"], [src="${url}"]`)) {
          return resolve(); // If already loaded, resolve immediately
        }

        const element = document.createElement(type === 'css' ? 'link' : 'script');

        if (type === 'css') {
          element.rel = 'stylesheet';
          element.href = url;
        } else {
          element.src = url;
          element.onload = resolve; // Resolve when script is loaded
        }

        document.head.appendChild(element);
        if (type === 'css') resolve(); // Resolve immediately for CSS
      });
    }

    async function initGallerySwiper() {
