  <script>
    // Universal Multi-Tabs by Rana
    (function () {
      function initSKBUniversalMultiTabs(scope = document) {
        const containers = scope.querySelectorAll('.skb_universal_tabs_wrap');

        containers.forEach((container) => {
          const tabs = container.querySelectorAll('[data-skb-tab-button]');
          const contents = container.querySelectorAll('[data-skb-tab-content]');
          const indicator = container.querySelector('.skb_tab_nav__position'); // Optional

          // Move indicator under active tab
          function moveIndicator(tab) {
            if (!indicator) return; // Skip if indicator is not found
            const tabRect = tab.getBoundingClientRect();
            const parentRect = tab.parentElement.getBoundingClientRect();
            const leftOffset = tabRect.left - parentRect.left;
            indicator.style.width = `${tabRect.width}px`;
            indicator.style.transform = `translateX(${leftOffset}px)`;
          }

          // Set initial state
          const firstActiveTab = container.querySelector('.skb_tab.skb_active') || tabs[0];
          if (firstActiveTab) {
            firstActiveTab.classList.add('skb_active');
            moveIndicator(firstActiveTab);
            const firstTarget = firstActiveTab.getAttribute('data-skb-tab-button');
            const firstContent = container.querySelector(`[data-skb-tab-content="${firstTarget}"]`);
            if (firstContent) {
              firstContent.classList.add('skb_show');
              firstContent.style.position = 'relative';
            }
          }

          // Add click events
          tabs.forEach((tab) => {
            tab.addEventListener('click', () => {
              const target = tab.getAttribute('data-skb-tab-button');

              tabs.forEach((t) => t.classList.remove('skb_active'));
              tab.classList.add('skb_active');

              moveIndicator(tab);

              contents.forEach((content) => {
                content.classList.remove('skb_show');
                content.style.position = 'absolute';
              });

              const next = container.querySelector(`[data-skb-tab-content="${target}"]`);
              if (next) {
                next.classList.add('skb_show');
                next.style.position = 'relative';
              }
            });
          });

          // Recalculate on window resize
          if (indicator) {
            window.addEventListener('resize', () => {
              const activeTab = container.querySelector('.skb_tab.skb_active');
              if (activeTab) moveIndicator(activeTab);
            });
          }
        });
      }

      // DOM ready
      document.addEventListener('DOMContentLoaded', () => {
        initSKBUniversalMultiTabs();
      });

      // Shopify section reload support
      if (typeof Shopify !== 'undefined' && Shopify.designMode) {
        document.addEventListener('shopify:section:load', (e) => {
          initSKBUniversalMultiTabs(e.target);
        });
      }
    })();

    document.addEventListener('DOMContentLoaded', function () {
      const tabsContainer = document.querySelector('.skb_tabs');
      const tabButtons = tabsContainer.querySelectorAll('.skb_tab');

      // Observe click events on each tab
      tabButtons.forEach((tab) => {
        tab.addEventListener('click', () => {
          setTimeout(() => {
            scrollActiveTabIntoView();
          }, 50); // Give some time for .skb_active to apply
        });
      });

      function scrollActiveTabIntoView() {
        const activeTab = tabsContainer.querySelector('.skb_tab.skb_active');
        if (!activeTab) return;

        const containerRect = tabsContainer.getBoundingClientRect();
        const tabRect = activeTab.getBoundingClientRect();

        const containerLeft = tabsContainer.scrollLeft;
        const offsetLeft = activeTab.offsetLeft;
        const offsetRight = offsetLeft + activeTab.offsetWidth;

        const visibleLeft = containerLeft;
        const visibleRight = containerLeft + tabsContainer.clientWidth;

        // Scroll left if tab is out of view to the left
        if (offsetLeft < visibleLeft) {
          tabsContainer.scrollTo({
            left: offsetLeft - 40, // Add some margin
            behavior: 'smooth',
          });
        }
        // Scroll right if tab is out of view to the right

<script>
  // Universal Multi-Tabs by Rana
  (function () {
    function initSKBUniversalMultiTabs(scope = document) {
      const containers = scope.querySelectorAll('.skb_universal_tabs_wrap');

      containers.forEach((container) => {
        const tabs = container.querySelectorAll('[data-skb-tab-button]');
        const contents = container.querySelectorAll('[data-skb-tab-content]');
        const indicator = container.querySelector('.skb_tab_nav__position'); // Optional

        const tabsWrapper = container.querySelector('.skb_tabs');

        // Move indicator under active tab
        function moveIndicator(tab) {
          if (!indicator || !tab || !tabsWrapper) return;

          const tabOffsetLeft = tab.offsetLeft;
          const tabWidth = tab.offsetWidth;

          indicator.style.width = `${tabWidth}px`;
          indicator.style.transform = `translateX(${tabOffsetLeft}px)`;
        }

        // Set initial state
        const firstActiveTab = container.querySelector('.skb_tab.skb_active') || tabs[0];
        if (firstActiveTab) {
          firstActiveTab.classList.add('skb_active');
          moveIndicator(firstActiveTab);
          const firstTarget = firstActiveTab.getAttribute('data-skb-tab-button');
          const firstContent = container.querySelector(`[data-skb-tab-content="${firstTarget}"]`);
          if (firstContent) {
            firstContent.classList.add('skb_show');
            firstContent.style.position = 'relative';
          }
        }

        // Add click events
        tabs.forEach((tab) => {
          tab.addEventListener('click', () => {
            const target = tab.getAttribute('data-skb-tab-button');

            tabs.forEach((t) => t.classList.remove('skb_active'));
            tab.classList.add('skb_active');

            moveIndicator(tab);
            scrollActiveTabIntoView(tab);

            contents.forEach((content) => {
              content.classList.remove('skb_show');
              content.style.position = 'absolute';
            });

            const next = container.querySelector(`[data-skb-tab-content="${target}"]`);
            if (next) {
              next.classList.add('skb_show');
              next.style.position = 'relative';
            }
          });
        });

        // Recalculate on window resize or scroll
        if (indicator) {
          window.addEventListener('resize', () => {
            const activeTab = container.querySelector('.skb_tab.skb_active');
            if (activeTab) moveIndicator(activeTab);
          });

          // Sync position while scrolling horizontally
          if (tabsWrapper) {
            tabsWrapper.addEventListener('scroll', () => {
              const activeTab = container.querySelector('.skb_tab.skb_active');
              if (activeTab) moveIndicator(activeTab);
            });
          }
        }

        // Scroll tab into view if needed
        function scrollActiveTabIntoView(tab) {
          if (!tabsWrapper || !tab) return;

          const containerLeft = tabsWrapper.scrollLeft;
          const offsetLeft = tab.offsetLeft;
          const offsetRight = offsetLeft + tab.offsetWidth;

          const visibleLeft = containerLeft;
          const visibleRight = containerLeft + tabsWrapper.clientWidth;

          if (offsetLeft < visibleLeft) {
            tabsWrapper.scrollTo({
              left: offsetLeft - 40,
              behavior: 'smooth',
            });
          } else if (offsetRight > visibleRight) {
            tabsWrapper.scrollTo({
              left: offsetRight - tabsWrapper.clientWidth + 40,
              behavior: 'smooth',
            });
          }
        }
      });
    }

    // DOM ready
    document.addEventListener('DOMContentLoaded', () => {
      initSKBUniversalMultiTabs();
    });

    // Shopify section reload support
    if (typeof Shopify !== 'undefined' && Shopify.designMode) {
      document.addEventListener('shopify:section:load', (e) => {
        initSKBUniversalMultiTabs(e.target);
      });
    }
  })();
</script>
