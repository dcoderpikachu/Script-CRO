<script>
  // SKB Universal Multi-Tabs
  (function () {
    function initSKBTabs(scope = document) {
      const tabContainers = scope.querySelectorAll('.skbtab-container');

      tabContainers.forEach((container) => {
        if (container.dataset.initialized === 'true') return;
        container.dataset.initialized = 'true';

        const tabButtons = container.querySelectorAll('.skb-tab-btn');
        const tabContents = container.querySelectorAll('.skb-tab-content');
        const contentsWrapper = container.querySelector('.skb-tab-contents');
        const tabsWrap = container.querySelector('.skb-tabs-wrap');
        const indicator = tabsWrap?.querySelector('.skb-tabs-nav__position');

        if (!tabButtons.length || !tabContents.length || !tabsWrap || !contentsWrapper) return;

        function updateIndicator(activeBtn) {
          if (!indicator || !activeBtn) return;

          const btnRect = activeBtn.getBoundingClientRect();
          const wrapRect = tabsWrap.getBoundingClientRect();
          const offsetX = btnRect.left - wrapRect.left;
          const width = btnRect.width;

          indicator.style.width = `${width}px`;
          indicator.style.transform = `translateX(${offsetX}px)`;
        }

        function switchTab(index) {
          const newActiveContent = tabContents[index];
          const currentActiveContent = container.querySelector('.skb-tab-content.active');

          if (currentActiveContent === newActiveContent) return;

          contentsWrapper.style.height = `${newActiveContent.scrollHeight}px`;

          if (currentActiveContent) {
            currentActiveContent.classList.remove('active');
            currentActiveContent.style.opacity = '0';
            currentActiveContent.style.visibility = 'hidden';
            currentActiveContent.style.position = 'absolute';
          }

          setTimeout(() => {
            requestAnimationFrame(() => {
              newActiveContent.classList.add('active');
              newActiveContent.style.position = 'relative';
              newActiveContent.style.visibility = 'visible';
              newActiveContent.style.opacity = '1';

              setTimeout(() => {
                contentsWrapper.style.height = 'auto';
              }, 200);
            });
          }, 200);
        }

        tabButtons.forEach((btn, index) => {
          btn.addEventListener('click', () => {
            tabButtons.forEach((b) => b.classList.remove('active'));
            btn.classList.add('active');

            switchTab(index);
            updateIndicator(btn);
          });
        });

        // Init state
        tabButtons[0]?.classList.add('active');
        tabContents[0]?.classList.add('active', 'initial-load');
        tabContents[0].style.position = 'relative';
        tabContents[0].style.opacity = '1';
        tabContents[0].style.visibility = 'visible';
        contentsWrapper.style.height = `${tabContents[0].scrollHeight}px`;

        updateIndicator(tabButtons[0]);

        setTimeout(() => {
          tabContents[0]?.classList.remove('initial-load');
        }, 10);
      });
    }

    // Run on DOM ready
    document.addEventListener('DOMContentLoaded', () => {
      initSKBTabs();
    });

    // Re-run on Shopify section load (in editor/design mode)
    if (typeof Shopify !== 'undefined' && Shopify.designMode) {
      document.addEventListener('shopify:section:load', (e) => {
        const section = e.target;
        initSKBTabs(section);
      });
    }
  })();

  // SKB Universal Collapsible
 (function () {
    function openContent(element) {
      element.style.overflow = 'hidden';
      const height = element.scrollHeight;

      const keyframes = {
        height: ['0px', `${height}px`],
        visibility: ['hidden', 'visible'],
        opacity: element.hasAttribute('data-animate-items') ? [0, 0] : [1, 1],
      };

      const animation = element.animate(keyframes, {
        duration: 500,
        direction: 'normal',
        easing: 'cubic-bezier(0.75, 0, 0.175, 1)',
        fill: 'forwards',
      });

      animation.onfinish = () => {
        element.style.overflow = 'visible';
        element.style.height = `${height}px`;
        element.setAttribute('data-open', '');
      };

      if (element.hasAttribute('data-animate-items')) {
        element.animate(
          {
            opacity: [0, 1],
            transform: ['translateY(10px)', 'translateY(0)'],
          },
          {
            duration: 250,
            delay: 250,
            easing: 'cubic-bezier(0.75, 0, 0.175, 1)',
            fill: 'forwards',
          }
        );
      }
    }

    function closeContent(element) {
      const height = element.scrollHeight;
      element.style.overflow = 'hidden';

      const animation = element.animate(
        {
          height: [`${height}px`, '0px'],
          visibility: ['visible', 'hidden'],
          opacity: element.hasAttribute('data-animate-items') ? [0, 0] : [1, 0],
        },
        {
          duration: 500,
          direction: 'normal',
          easing: 'cubic-bezier(0.75, 0, 0.175, 1)',
          fill: 'forwards',
        }
      );

      animation.onfinish = () => {
        element.style.overflow = 'hidden';
        element.removeAttribute('data-open');
      };
    }

    function initCollapsibleToggle(scope = document) {
      const wraps = scope.querySelectorAll('ed-collapsible-wrap');

      wraps.forEach((wrap) => {
        const toggles = wrap.querySelectorAll('.ed-collapsible-toggle');

        toggles.forEach((toggle) => {
          if (toggle.dataset.initialized === 'true') return;

          toggle.dataset.initialized = 'true';

          const contentId = toggle.getAttribute('aria-controls');
          const content = wrap.querySelector(`#${contentId}`);
          if (!content) return;

          // Keep initial state as-is (open or closed)
          const isInitiallyOpen = content.hasAttribute('data-open');
          toggle.setAttribute('aria-expanded', String(isInitiallyOpen));
          toggle.classList.toggle('is-open', isInitiallyOpen);

          toggle.addEventListener('click', () => {
            const isOpen = toggle.getAttribute('aria-expanded') === 'true';

            toggle.setAttribute('aria-expanded', String(!isOpen));
            toggle.classList.toggle('is-open', !isOpen);

            if (!isOpen) {
              openContent(content);
            } else {
              closeContent(content);
            }
          });
        });
      });
    }

    // Initialize when DOM is ready
    document.addEventListener('DOMContentLoaded', function () {
      initCollapsibleToggle();
    });

    // Reinitialize for each section dynamically loaded
    if (typeof Shopify !== 'undefined' && Shopify.designMode) {
      document.addEventListener('shopify:section:load', function (e) {
        const section = e.target;
        initCollapsibleToggle(section);
      });
    }
  })();
</script>
