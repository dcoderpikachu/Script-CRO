<script>
  (function () {
    function initMultipleImageSlideSwiper() {
      const contentSwiper = new Swiper('.contentSwiper__{{ section.id }}', {
        slidesPerView: 1,
        loop: true,
        allowTouchMove: true,
        autoHeight: true,
        effect: 'fade',
        navigation: {
          nextEl: '.contentSwiper_next',
          prevEl: '.contentSwiper_prev',
        },
      });

      const mediaSwiper = new Swiper('.mediaSwiper__{{ section.id }}', {
        slidesPerView: 1,
        spaceBetween: 30,
        effect: 'fade',
        loop: true,
        navigation: false,
      });

      // Sync the two swipers
      mediaSwiper.controller.control = contentSwiper;
      contentSwiper.controller.control = mediaSwiper;
    }

    // Run the function to load resources and initialize Swiper
    initMultipleImageSlideSwiper();

    if (typeof Shopify !== 'undefined' && Shopify.designMode) {
      document.addEventListener('shopify:section:load', function () {
        initMultipleImageSlideSwiper();
      });
    }
  })();
</script>
