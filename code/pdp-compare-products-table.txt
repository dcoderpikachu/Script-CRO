<script>
  function renderCompareTable() {
    const wrapper = document.getElementById('compare-products-wrapper');
    const compareMain = document.querySelector('.ed_compare__main');
    const compareChart = document.querySelector('.ed_compare_chart');
    const table = document.getElementById('dynamic-spec-rows');
    const compareList = JSON.parse(localStorage.getItem('compareList')) || [];

    // Clear previous content to avoid duplication
    if (wrapper) wrapper.innerHTML = '';
    if (table) table.innerHTML = '';

    // Set CSS variable for number of products
    if (compareMain) {
      compareMain.style.setProperty('--number-of-product', compareList.length);
    }

    if (!wrapper || compareList.length === 0 || !compareChart) {
      if (compareMain) compareMain.classList.add('ed_empty_compare_list');
      return;
    }

    // STEP 1: Render product cards
    compareList.forEach((product) => {
      let ed_price_currency = '{{ shop.currency }}';
      let price = product.price / 100;

      // Format the price based on the currency
      let formattedPrice;

      if (ed_price_currency === 'EUR') {
        formattedPrice = '€' + price.toFixed(2).replace('.', ',');
      } else {
        formattedPrice = '$' + price.toFixed(2);
      }

      const productHTML = `
        <div class="ed_compare_chart__product">
          <a href="/products/${product.handle}">
            <img src="${product.featured_image}" alt="${product.title}">
          </a>
          <div class="ed_compare_chart__product-details">
            <div class="ed_compare_chart__product-info">
              <div class="ed_compare_chart__product-title">
                <a href="/products/${product.handle}">${product.title}</a>
              </div>
              <div class="ed_compare_chart__product-price">
                <span>${formattedPrice}</span>
              </div>
            </div>
            <div class="ed_compare_chart__product-button">
              <a href="/products/${product.handle}" class="button button--primary ed_compare_chart__product-btn ed-btn-rounded">View</a>
              <button type="button" class="clear-compare" data-id="${product.id}">
                <svg xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" stroke-width="1.5" stroke="currentColor" class="size-6">
                  <path stroke-linecap="round" stroke-linejoin="round" d="m9.75 9.75 4.5 4.5m0-4.5-4.5 4.5M21 12a9 9 0 1 1-18 0 9 9 0 0 1 18 0Z" />
                </svg>
              </button>
            </div>
          </div>
        </div>
      `;
      wrapper.insertAdjacentHTML('beforeend', productHTML);
    });

    // STEP 2: Render spec comparison rows
    const rowNamesString = compareChart.dataset.tableRowNames || '';
    const rowNames = rowNamesString
      .split('|')
      .map((name) => name.trim())
      .filter(Boolean);

    rowNames.forEach((rowName, index) => {
      const row = document.createElement('div');
      row.classList.add('ed_compare_chart__table-row');

      const heading = document.createElement('div');
      heading.classList.add('ed_compare_chart__heading');
      heading.textContent = rowName;
      row.appendChild(heading);

      compareList.forEach((product) => {
        const fieldKey = `field_${index + 1}`;
        const value = product.metafields?.[fieldKey] || '-';

        const valueDiv = document.createElement('div');
        valueDiv.classList.add('ed_compare_chart__value');
        valueDiv.innerHTML = `<p>${value}</p>`;
        row.appendChild(valueDiv);
      });

      table.appendChild(row);
    });

    // STEP 3: Clear product from compare list
    const clearCompareButtons = document.querySelectorAll('.clear-compare');
    clearCompareButtons.forEach((btn) => {
      btn.addEventListener('click', function (e) {
        e.preventDefault();
        const idToRemove = this.dataset.id;
        let updatedList = compareList.filter((product) => product.id != idToRemove);
        localStorage.setItem('compareList', JSON.stringify(updatedList));
        location.reload();
      });
    });
  }

  // Run once on DOM load
  document.addEventListener('DOMContentLoaded', function () {
    renderCompareTable();
  });

  // Re-run on Shopify updates
  if (typeof Shopify !== 'undefined' && Shopify.designMode) {
    document.addEventListener('shopify:section:load', function () {
      renderCompareTable();
    });
  }
</script>

<script>
    function initCompareButtons() {
        const compareButtons = document.querySelectorAll('.ed_btn_add_product_to_compare_section');

        if (!compareButtons.length) return;

        compareButtons.forEach((button) => {
          // Parse base product data
          const product = JSON.parse(button.dataset.product);

          // Step 1: Read and parse metafields string (dynamic keys)
          const rawMetafields = button.dataset.productMetafields || '';
          const metafieldsArray = rawMetafields
            .split('|')
            .map((item) => item.trim())
            .filter(Boolean);

          // Step 2: Create dynamic keys like field_1, field_2, ...
          const metafieldsObject = {};
          metafieldsArray.forEach((value, index) => {
            metafieldsObject[`field_${index + 1}`] = value;
          });

          // Step 3: Attach parsed metafields to product
          product.metafields = metafieldsObject;

          const compareUrl = button.dataset.compareUrl;
          let compareList = JSON.parse(localStorage.getItem('compareList')) || [];

          // Step 4: Update UI if already in compare
          if (compareList.some((p) => p.id === product.id)) {
            button.textContent = 'View at compare';
            button.setAttribute('href', compareUrl);
            button.classList.add('is-in-compare');
          }

          // Step 5: Add click event to add to compare list
          button.addEventListener('click', function (e) {
            e.preventDefault();

            let compareList = JSON.parse(localStorage.getItem('compareList')) || [];

            const isAlreadyAdded = compareList.some((p) => p.id === product.id);

            if (!isAlreadyAdded) {
              compareList.push(product);
              localStorage.setItem('compareList', JSON.stringify(compareList));
              button.textContent = 'Compare now';
              button.setAttribute('href', compareUrl);
              button.classList.add('is-in-compare');
            } else {
              // Redirect if already added
              window.location.href = compareUrl;
            }
          });
        });
    }

    document.addEventListener('DOMContentLoaded', initCompareButtons);

    document.addEventListener('collection:updated', initCompareButtons);

    const observerCompare = new MutationObserver((mutations) => {
        for (const mutation of mutations) {
          if (mutation.addedNodes.length) {
            const productListChanged = [...mutation.addedNodes].some(
              (node) => node.nodeType === 1 && node.querySelector?.('.product-item')
            );
            if (productListChanged) {
              initCompareButtons();
              break;
            }
          }
        }
    });

    observerCompare.observe(document.body, {
        childList: true,
        subtree: true,
    });
</script>
