<script>
   function initSwiperFiles() {
    // Initialize Swiper after script is loaded
    const leftSwiper = new Swiper('.leftSwiper', {
      loop: true,
      effect: 'fade',
      spaceBetween: 1,
      allowTouchMove: false,
      navigation: {
        nextEl: '.leftSwiper_next',
        prevEl: '.leftSwiper_prev',
      },
      pagination: {
        el: '.leftSwiper_pagination',
        clickable: true,
      },
    });

    const rightSwiper = new Swiper('.rightSwiper', {
      loop: true,
      spaceBetween: 1,
      navigation: false,
    });

    // Sync the two swipers
    leftSwiper.controller.control = rightSwiper;
    rightSwiper.controller.control = leftSwiper;
  }

  document.addEventListener('DOMContentLoaded', function () {
    initSwiperFiles();
  });

  if (typeof Shopify !== 'undefined' && Shopify.designMode) {
    document.addEventListener('shopify:section:load', function () {
      initSwiperFiles();
    });
  }
</script>
