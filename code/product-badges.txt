Step 2: Create snippet product-labels.liquid
{%- capture product_labels -%}
  {%- for tag in product.tags -%}
    {%- if tag contains '__label' -%}
      {% assign label = tag | split: ':' | last %}

      {% for entity in shop.metaobjects.product_badge.values %}
        {% if label == entity.badge_name %}
          {% assign label_color = entity.color %}
          {% assign label_bg_color = entity.background_color %}
        {% endif %}
      {% endfor %}

      <span style="{% if label_color and label_bg_color %} color:{{ label_color }} !important; background-color:{{ label_bg_color }} !important; {% endif %}" 
            class="label label--custom{% if tag contains '__label2' %}2{% endif %}">
        {{ tag | split: ':' | last }}
      </span>
      {% assign label_color = "" %}
       {% assign label_bg_color = "" %}
    {%- endif -%}
  {%- endfor -%}

  {%- unless product.available -%}
    <span class="label label--subdued">{{ 'collection.product.sold_out' | t }}</span>
  {%- endunless -%}

  {%- assign cheapest_variant = product.variants | sort: 'price' | first -%}

  {%- if settings.show_discount and product.available and product.price < product.compare_at_price and cheapest_variant.compare_at_price != blank -%}
    {%- if settings.discount_mode == 'percentage' -%}
      {%- assign savings = product.compare_at_price | minus: product.price | times: 100.0 | divided_by: product.compare_at_price | round | append: '%' -%}
    {%- else -%}
      {%- capture savings -%}{{ product.compare_at_price | minus: product.price | money }}{%- endcapture -%}
    {%- endif -%}

    <span class="label label--highlight">{{ 'collection.product.discount_html' | t: savings: savings }}</span>
  {%- endif -%}
{%- endcapture -%}

{%- if product_labels != blank and reduced_content != true -%}
  <div class="product-item__label-list label-list">
    {{- product_labels -}}
  </div>
{%- endif -%}
