<script>
class UniversalStickyATC {
  constructor() {
    this.stickyATC = document.querySelector(".universal-sticky-atc");
    if (!this.stickyATC) return;

    this.productId = this.stickyATC.dataset.productId;
    this.originalForm = document.getElementById(`{{ product_form_id }}`);
    this.paymentContainer = document.getElementById(
      "{{ payment_container_id }}"
    );

    this.init();
    this.setupVariantSelectors();
    this.setupDropdowns();
  }

  init() {
    this.setupEventListeners();
    this.setupIntersectionObserver();
    this.updateStickyPosition();
    window.addEventListener("resize", () => this.updateStickyPosition());
  }

  updateStickyPosition() {
    const header = document.querySelector("{{ header_selector }}");
    if (header && this.stickyATC) {
      const headerBottom =
        header.getBoundingClientRect().bottom + window.scrollY;
      this.stickyATC.style.setProperty(
        "--sticky-top-position",
        `${headerBottom}px`
      );
    }
  }

  setupDropdowns() {
    const dropdownToggles = this.stickyATC.querySelectorAll(
      ".sticky-atc-dropdown-toggle"
    );

    dropdownToggles.forEach((toggle) => {
      toggle.addEventListener("click", (e) => {
        e.stopPropagation();
        const dropdown = toggle.closest(".sticky-atc-dropdown");
        const menu = dropdown.querySelector(".sticky-atc-dropdown-menu");

        document.querySelectorAll(".sticky-atc-dropdown-menu").forEach((m) => {
          if (m !== menu) m.classList.remove("show");
        });

        menu.classList.toggle("show");

        const icon = toggle.querySelector("svg");
        if (menu.classList.contains("show")) {
          icon.style.transform = "rotate(180deg)";
        } else {
          icon.style.transform = "rotate(0)";
        }
      });
    });

    document.addEventListener("click", () => {
      this.stickyATC
        .querySelectorAll(".sticky-atc-dropdown-menu")
        .forEach((menu) => {
          menu.classList.remove("show");
        });
      this.stickyATC
        .querySelectorAll(".sticky-atc-dropdown-toggle svg")
        .forEach((icon) => {
          icon.style.transform = "rotate(0)";
        });
    });
  }

  setupVariantSelectors() {
    const colorSwatches = this.stickyATC.querySelectorAll(
      ".sticky-atc-color-swatch .swatch"
    );
    colorSwatches.forEach((swatch) => {
      swatch.addEventListener("click", () => {
        const container = swatch.closest(".sticky-atc-color-swatch");
        container
          .querySelectorAll(".swatch")
          .forEach((s) => s.classList.remove("active"));
        swatch.classList.add("active");
        this.handleVariantChange();
      });
    });

    const dropdownItems = this.stickyATC.querySelectorAll(
      ".sticky-atc-dropdown-item"
    );
    dropdownItems.forEach((item) => {
      item.addEventListener("click", () => {
        const dropdown = item.closest(".sticky-atc-dropdown");
        const menu = dropdown.querySelector(".sticky-atc-dropdown-menu");
        const toggle = dropdown.querySelector(".sticky-atc-dropdown-toggle");
        const selectedValue = dropdown.querySelector(
          ".sticky-atc-dropdown-selected"
        );

        menu
          .querySelectorAll(".sticky-atc-dropdown-item")
          .forEach((i) => i.classList.remove("active"));
        item.classList.add("active");

        selectedValue.textContent = item.dataset.value;

        menu.classList.remove("show");
        toggle.querySelector("svg").style.transform = "rotate(0)";

        this.handleVariantChange();
      });
    });
  }

  handleVariantChange() {
    const selectedOptions = this.getSelectedOptions();
    const variant = this.findVariant(selectedOptions);

    if (variant) {
      this.updateVariantInfo(variant);
      this.updateSoldOutStates(selectedOptions);
      this.dispatchVariantChange(variant);
    }
  }

  getSelectedOptions() {
    const selectedOptions = [];
    const optionContainers = this.stickyATC.querySelectorAll(
      ".swatch-container, .sticky-atc-dropdown-container"
    );

    optionContainers.forEach((container) => {
      if (container.classList.contains("sticky-atc-color-swatch")) {
        const activeSwatch = container.querySelector(".swatch.active");
        selectedOptions.push(activeSwatch ? activeSwatch.dataset.value : null);
      } else {
        const activeItem = container.querySelector(
          ".sticky-atc-dropdown-item.active"
        );
        selectedOptions.push(activeItem ? activeItem.dataset.value : null);
      }
    });

    return selectedOptions;
  }

  findVariant(selectedOptions) {
    const variants = JSON.parse(
      document.querySelector(`#product-${this.productId}`).dataset.variants
    );

    return variants.find((variant) => {
      return (
        variant.option1 === selectedOptions[0] &&
        (!selectedOptions[1] || variant.option2 === selectedOptions[1]) &&
        (!selectedOptions[2] || variant.option3 === selectedOptions[2])
      );
    });
  }

  updateSoldOutStates(selectedOptions) {
    const variants = JSON.parse(
      document.querySelector(`#product-${this.productId}`).dataset.variants
    );
    const optionContainers = this.stickyATC.querySelectorAll(
      ".swatch-container, .sticky-atc-dropdown-container"
    );

    optionContainers.forEach((container, index) => {
      let items;

      if (container.classList.contains("sticky-atc-color-swatch")) {
        items = container.querySelectorAll(".swatch");
      } else {
        items = container.querySelectorAll(".sticky-atc-dropdown-item");
      }

      items.forEach((item) => {
        const testOptions = [...selectedOptions];
        testOptions[index] = item.dataset.value;

        const variant = this.findVariant(testOptions);
        if (variant && !variant.available) {
          item.classList.add("soldout_variant");
        } else {
          item.classList.remove("soldout_variant");
        }
      });
    });
  }

  setupEventListeners() {
    document.addEventListener("variant:changed", (event) => {
      // only for focal theme
      this.updateVariantInfo(event.detail.variant);
      this.syncSelectorsWithVariant(event.detail.variant);
    });

    const addToCartButton = this.stickyATC.querySelector("[data-add-to-cart]");
    if (addToCartButton) {
      addToCartButton.addEventListener("click", () => {
        if (this.originalForm) {
          this.originalForm.dispatchEvent(
            new Event("submit", { bubbles: true })
          );
        }
      });
    }
  }

  syncSelectorsWithVariant(variant) {
    const optionContainers = this.stickyATC.querySelectorAll(
      ".swatch-container, .sticky-atc-dropdown-container"
    );

    optionContainers.forEach((container, index) => {
      const optionValue = variant[`option${index + 1}`];

      if (container.classList.contains("sticky-atc-color-swatch")) {
        const swatches = container.querySelectorAll(".swatch");
        swatches.forEach((swatch) => {
          swatch.classList.remove("active");
          if (swatch.dataset.value === optionValue) {
            swatch.classList.add("active");
          }
        });
      } else {
        const items = container.querySelectorAll(".sticky-atc-dropdown-item");
        const selectedValue = container.querySelector(
          ".sticky-atc-dropdown-selected"
        );

        items.forEach((item) => {
          item.classList.remove("active");
          if (item.dataset.value === optionValue) {
            item.classList.add("active");
            selectedValue.textContent = optionValue;
          }
        });
      }
    });
  }

  dispatchVariantChange(variant) {
    const variantId = variant.id;
    const newUrl = new URL(window.location.href);
    if (variantId) {
      newUrl.searchParams.set("variant", variantId);
    } else {
      newUrl.searchParams.delete("variant");
    }
    window.history.replaceState(
      { path: newUrl.toString() },
      "",
      newUrl.toString()
    );
    const idInput = this.originalForm.querySelector('[name="id"]');
    if (idInput) {
      idInput.value = variantId;
    }

    // only for focal theme
    this.originalForm.dispatchEvent(
      new CustomEvent("variant:changed", {
        bubbles: true,
        detail: {
          variant: variant,
        },
      })
    );
    const variantSelector = document.querySelector("product-variants");
    variantSelector.selectVariant(variantId);
    variantSelector._onMasterSelectorChanged();
  }

  setupIntersectionObserver() {
    if (!this.paymentContainer || !this.stickyATC) {
      console.warn("Missing required elements for sticky ATC");
      return;
    }

    let lastScrollY = window.scrollY;
    let hasUserScrolled = false;
    let isPaymentContainerPassed = false;

    const paymentContainerOffset =
      this.paymentContainer.offsetTop + this.paymentContainer.offsetHeight;

    window.addEventListener("scroll", () => {
      const currentScrollY = window.scrollY;
      const isScrollingDown = currentScrollY > lastScrollY;
      lastScrollY = currentScrollY;

      if (!hasUserScrolled && currentScrollY > 0) {
        hasUserScrolled = true;
      }

      const isContainerPassed = currentScrollY > paymentContainerOffset;

      isPaymentContainerPassed = isContainerPassed;

      const shouldShowStickyATC = hasUserScrolled && isPaymentContainerPassed;

      if (shouldShowStickyATC) {
        this.stickyATC.classList.add("active");
      } else {
        this.stickyATC.classList.remove("active");
      }
    });
  }

  updateVariantInfo(variant) {
    if (!variant) return;

    const priceContainer = this.stickyATC.querySelector(
      "[data-price-container]"
    );
    if (priceContainer) {
      const priceHTML = this.formatMoney(variant.price);
      let priceContent = `<span class="sticky-atc-current-price">${priceHTML}</span>`;

      if (variant.compare_at_price > variant.price) {
        const comparePriceHTML = this.formatMoney(variant.compare_at_price);
        const savings = variant.compare_at_price - variant.price;
        priceContent += `
          <span class="sticky-atc-compare-price">${comparePriceHTML}</span>
          <span class="sticky-atc-sale-badge">{{ savings_label_text }} ${this.formatMoney(
            savings
          )}</span>
        `;
      }

      priceContainer.innerHTML = priceContent;
    }
    const addToCartButton = this.stickyATC.querySelector("[data-add-to-cart]");
    if (addToCartButton) {
      if (variant.available) {
        addToCartButton.disabled = false;
        addToCartButton.textContent = "{{atc_button_text}}";
      } else {
        addToCartButton.disabled = true;
        addToCartButton.textContent = "{{atc_button_soldout}}";
      }
    }

    if (variant.featured_media) {
      const imageElement = this.stickyATC.querySelector(".sticky-atc-image");
      if (imageElement) {
        imageElement.src = this.getSizedImageUrl(
          variant.featured_media,
          "165x"
        );
        imageElement.srcset = this.getImageSrcset(
          variant.featured_media,
          [55, 110, 165]
        );
        imageElement.alt = variant.featured_media.alt || "";
      }
    }
  }

  formatMoney(cents, format) {
    if (typeof cents === "string") {
      cents = cents.replace(".", "");
    }
    const placeholderRegex = /\{\{\s*(\w+)\s*\}\}/,
      formatString = format || window.themeVariables.settings.moneyFormat;
    function defaultTo(value2, defaultValue) {
      return value2 == null || value2 !== value2 ? defaultValue : value2;
    }
    function formatWithDelimiters(number, precision, thousands, decimal) {
      precision = defaultTo(precision, 2);
      thousands = defaultTo(thousands, ",");
      decimal = defaultTo(decimal, ".");
      if (isNaN(number) || number == null) {
        return 0;
      }
      number = (number / 100).toFixed(precision);
      let parts = number.split("."),
        dollarsAmount = parts[0].replace(
          /(\d)(?=(\d\d\d)+(?!\d))/g,
          "$1" + thousands
        ),
        centsAmount = parts[1] ? decimal + parts[1] : "";
      return dollarsAmount + centsAmount;
    }
    let value = "";
    switch (formatString.match(placeholderRegex)[1]) {
      case "amount":
        value = formatWithDelimiters(cents, 2);
        break;
      case "amount_no_decimals":
        value = formatWithDelimiters(cents, 0);
        break;
      case "amount_with_space_separator":
        value = formatWithDelimiters(cents, 2, " ", ".");
        break;
      case "amount_with_comma_separator":
        value = formatWithDelimiters(cents, 2, ".", ",");
        break;
      case "amount_with_apostrophe_separator":
        value = formatWithDelimiters(cents, 2, "'", ".");
        break;
      case "amount_no_decimals_with_comma_separator":
        value = formatWithDelimiters(cents, 0, ".", ",");
        break;
      case "amount_no_decimals_with_space_separator":
        value = formatWithDelimiters(cents, 0, " ");
        break;
      case "amount_no_decimals_with_apostrophe_separator":
        value = formatWithDelimiters(cents, 0, "'");
        break;
    }
    if (formatString.indexOf("with_comma_separator") !== -1) {
      return formatString.replace(placeholderRegex, value);
    } else {
      return formatString.replace(placeholderRegex, value);
    }
  }

  getSizedImageUrl(media, size) {
    if (!media || !media.preview_image) return "";
    return (
      media.preview_image.src.split("?")[0] +
      `?width=${size.split("x")[0]}&height=${
        size.split("x")[1] || size.split("x")[0]
      }&crop=center`
    );
  }

  getImageSrcset(media, sizes) {
    if (!media || !media.preview_image) return "";
    return sizes
      .map((size) => {
        return `${this.getSizedImageUrl(media, `${size}x`)} ${size}w`;
      })
      .join(", ");
  }
}
