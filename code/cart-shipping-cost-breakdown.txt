function setCookie(cname, cvalue, exdays) {
    const d = new Date();
    d.setTime(d.getTime() + exdays * 24 * 60 * 60 * 1000);
    let expires = "expires=" + d.toUTCString();
    document.cookie = cname + "=" + cvalue + ";" + expires + ";path=/";
  }

  function getCookie(cname) {
    let name = cname + "=";
    let decodedCookie = decodeURIComponent(document.cookie);
    let ca = decodedCookie.split(";");
    for (let i = 0; i < ca.length; i++) {
      let c = ca[i];
      while (c.charAt(0) == " ") {
        c = c.substring(1);
      }
      if (c.indexOf(name) == 0) {
        return c.substring(name.length, c.length);
      }
    }
    return "";
  }

  class MiniShippingEstimator extends HTMLElement {
    connectedCallback() {
      this.select = this.querySelector(".country-select");
      this.zipInput = this.querySelector(".zip-code-input");
      this.zipSubmitBtn = this.querySelector(".zip-submit-btn");
      this.shippingError = this.querySelector(".shipping-estimate-error");
      this.shippingValue = this.querySelector(".country__shipping-value");

      this.customerLoggedIn =
        this.getAttribute("data-customer-logged-in") === "true";
      this.customerZip = this.getAttribute("data-customer-zip");
      this.customerCountry = this.getAttribute("data-customer-country");
      this.defaultCountry = "Germany";

      this.storedZip = this.getCookie("UserZip");
      this.storedCountry = this.getCookie("UserCountry") || this.defaultCountry;

      if (this.customerLoggedIn) {
        this.select.value =
          this.storedCountry || this.customerCountry || this.defaultCountry;
        this.zipInput.value = this.storedZip || this.customerZip || "";
      } else {
        this.select.value = this.storedCountry;
        this.zipInput.value = this.storedZip || "";
      }

      this.setCookie("UserCountry", this.select.value, 30);

      this.select.addEventListener("change", () => this.handleCountryChange());
      this.zipSubmitBtn.addEventListener("click", () =>
        this.saveZipAndEstimate()
      );

      this._estimateShipping();
    }

    handleCountryChange() {
      const selectedCountry = this.select.value;
      this.setCookie("UserCountry", selectedCountry, 30);

      this.zipInput.style.display = "block";
      this.zipSubmitBtn.style.display = "block";

      this.setCookie("UserZip", "", -1);
      this.zipInput.value = "";

      this.shippingValue.innerHTML = "";
    }

    saveZipAndEstimate() {
      const zip = this.zipInput.value.trim();
      if (!zip) return;

      this.setCookie("UserZip", zip, 30);
      this._estimateShipping();
    }

    async _estimateShipping() {
      const zip = this.zipInput.value.trim();
      const country = this.select.value;

      if (!zip) {
        this.shippingError.textContent =
          "Bitte geben Sie eine Postleitzahl ein.";
        this.shippingError.style.display = "block";
        return;
      }

      this.shippingError.style.display = "none";
      this.shippingValue.innerHTML = `<div class="spinner"><svg focusable="false" width="24" height="24" class="icon icon--spinner" viewBox="25 25 50 50"><circle cx="50" cy="50" r="20" fill="none" stroke="currentColor" stroke-width="5"></circle></svg></div>`;

      try {
        const response = await fetch(
          `${window.themeVariables.routes.cartUrl}/prepare_shipping_rates.json?shipping_address[zip]=${zip}&shipping_address[country]=${country}`,
          { method: "POST" }
        );

        if (response.ok) {
          const shippingRates = await this._getAsyncShippingRates(zip, country);
          this._formatShippingRates(shippingRates);
        } else {
          const jsonError = await response.json();
          this._formatError(jsonError);
          this.zipInput.value = "";
        }
      } catch (error) {
        this._formatError({
          error: "Fehler bei der Berechnung der Versandkosten.",
        });
        this.zipInput.value = "";
      }
    }

    async _getAsyncShippingRates(zip, country) {
      const response = await fetch(
        `${window.themeVariables.routes.cartUrl}/async_shipping_rates.json?shipping_address[zip]=${zip}&shipping_address[country]=${country}`
      );
      const responseText = await response.text();
      return responseText === "null"
        ? this._getAsyncShippingRates(zip, country)
        : JSON.parse(responseText).shipping_rates;
    }

    _formatShippingRates(shippingRates) {
      if (shippingRates.length === 0) {
        this.shippingValue.innerHTML = "Keine Versandoptionen verfügbar.";
        return;
      }

      const shippingPrice = parseFloat(shippingRates[0].price * 100);
      this.shippingValue.innerHTML =
        shippingPrice === 0
          ? `<span class="free-shipping-label">Kostenlos</span>`
          : `${formatMoney(shippingPrice)}`;
    }

    _formatError(errors) {
      this.shippingError.textContent =
        errors.error || "Fehler bei der Berechnung der Versandkosten.";
      this.shippingError.style.display = "block";
    }

    setCookie(name, value, days) {
      const d = new Date();
      d.setTime(d.getTime() + days * 24 * 60 * 60 * 1000);
      document.cookie = `${name}=${value};expires=${d.toUTCString()};path=/`;
    }

    getCookie(name) {
      const cookies = document.cookie.split("; ");
      for (const cookie of cookies) {
        const [key, value] = cookie.split("=");
        if (key === name) return value;
      }
      return "";
    }
  }

  window.customElements.define(
    "mini-shipping-estimator",
    MiniShippingEstimator
  );
