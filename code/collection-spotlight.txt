Step: 0
Prerequisites and safety
Duplicate your theme (Online Store → Themes → Actions → Duplicate) and work in the duplicate.
Use the Code Editor. After each change, preview the theme and check the browser console for errors.

Step: 1
Find the collection products section in your theme
If your theme uses JSON templates: open templates/collection.json and find the section that renders products. Note the section filename under sections/.
If your theme uses Liquid templates: open templates/collection.liquid (or collection.alternate.liquid) and find the section tag that renders the product grid. Open the referenced file in sections/.
Confirm the file contains a product loop like {% for product in collection.products %}.

Add block definitions to the section schema
In the same file, scroll to the {% schema %} ... {% endschema %} block.
Inside the JSON, find "blocks": [ ... ]. Paste the following block objects inside this array.
Important JSON rules: add a comma after the previous block, and do not add a trailing comma after the last block. Do not paste outside the "blocks" array:

    function initSpotlightVideos() {
      const videoItems = document.querySelectorAll('.spotlight_video__item');
      if (!videoItems.length) return;

      videoItems.forEach((container) => {
        // Skip if already initialized
        if (container.dataset.initialized === 'true') return;
        
        const playButton = container.querySelector('.spotlight_video_controls');
        const shadow = container.querySelector('.spotlight_video_shadow');
        const poster = container.querySelector('.spotlight_video__cover');
        const video = container.querySelector('video');

        if (!video) return;

        const hasAutoplay = video.hasAttribute('autoplay');

        const showControls = () => {
          if (playButton) playButton.style.display = '';
          if (shadow) shadow.style.display = '';
          if (poster) poster.style.display = '';
        };

        const hideControls = () => {
          if (playButton) playButton.style.display = 'none';
          if (shadow) shadow.style.display = 'none';
          if (poster) poster.style.display = 'none';
        };

        // Shared play logic
        const handlePlay = () => {
          video.play();
          container.classList.add('active');
          hideControls();
        };

        // Shared pause logic
        const handlePause = () => {
          video.pause();
          container.classList.remove('active');
          showControls();
        };

        // Autoplay case
        if (hasAutoplay) {
          // Hide controls initially for autoplay videos
          hideControls();
          container.classList.add('active');

          // Video click toggles pause and shows controls
          video.addEventListener('click', () => {
            if (!video.paused) {
              handlePause();
            }
          });

          // Play button click resumes playback
          if (playButton) {
            playButton.addEventListener('click', (e) => {
              e.stopPropagation();
              handlePlay();
            });
          }
        } else {
          // Non-autoplay: toggle play/pause with UI changes
          if (playButton) {
            playButton.addEventListener('click', (e) => {
              e.stopPropagation();
              handlePlay();
            });
          }

          video.addEventListener('click', () => {
            if (!video.paused) {
              handlePause();
            }
          });
        }

        // Mark as initialized
        container.dataset.initialized = 'true';
      });
    }

    // Run on DOMContentLoaded
    document.addEventListener('DOMContentLoaded', initSpotlightVideos);

    // Handle Shopify theme editor
    if (typeof Shopify !== 'undefined' && Shopify.designMode) {
      document.addEventListener('shopify:section:load', function () {
        initSpotlightVideos();
      });
    }

    // Handle AJAX pagination and other dynamic content updates
    // Listen for common AJAX events that might occur during pagination
    document.addEventListener('DOMNodeInserted', function(e) {
      if (e.target.querySelector && e.target.querySelector('.spotlight_video__item')) {
        setTimeout(initSpotlightVideos, 100);
      }
    });

    // Also listen for collection page updates (common in Shopify themes)
    document.addEventListener('collection:loaded', function() {
      setTimeout(initSpotlightVideos, 100);
    });

    // Listen for any content updates in the main content area
    const observer = new MutationObserver(function(mutations) {

      });
      if (shouldReinit) {

    });

    // Start observing
    observer.observe(document.body, {
