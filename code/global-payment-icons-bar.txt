        <div
          class="ed-payment-bar-cart cart-payment-bar"
          id="ed-payment-bar-cart"
          style=" background-color: {{settings.payment_bar_bg}}; {% if settings.payment_bar_border_radious == "rounded" %} border-radius: 10px; {% endif %}"
        >
          {% render 'ed-payment-bar' %}

            {% render 'ed-klarna-custom-payment' %}
          {% endif %}
        </div>
        {% unless settings.cart_payment_bar_show %}

            .cart-payment-bar{
              display: none;
            }
          </style>
        {% endunless %}
