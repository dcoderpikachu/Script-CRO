Step 3: Add this code in  assets/custom.js

function cpProductSlider(){
    const sliders = document.querySelectorAll("[data-slider]");
  sliders.forEach((slider) => {
    const sliderInner = slider.querySelector(".cp-product-slider-inner");
    if (sliderInner && sliderInner.children.length > 1) {
      const config = JSON.parse(sliderInner.getAttribute("data-config"));
      new Flickity(sliderInner, config);
    }
  });
}
cpProductSlider();
