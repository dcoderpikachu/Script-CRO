<script>
  function initEdTabsSection() {
    const tabContainers = document.querySelectorAll('.ed_tabs_section');

    tabContainers.forEach((container) => {
      const tabButtons = container.querySelectorAll('.ed_tabs__button');
      const tabContents = container.querySelectorAll('.ed_tabs__content');
      const tabThumbnails = container.querySelectorAll('.ed_tabs__thumbnail');

      tabButtons.forEach((btn, index) => {
        btn.addEventListener('click', function () {
          // Remove active classes from all buttons and contents in this container
          tabButtons.forEach((t) => t.classList.remove('active'));
          tabContents.forEach((c) => c.classList.remove('active'));
          tabThumbnails.forEach((c) => c.classList.remove('active'));

          // Add active class to the clicked tab and corresponding content
          btn.classList.add('active');
          tabContents[index].classList.add('active');
          tabThumbnails[index].classList.add('active');
        });
      });

      // Optionally activate the first tab by default
      tabButtons[0].classList.add('active');
      tabContents[0].classList.add('active');
      tabThumbnails[0].classList.add('active');
    });
  }

  document.addEventListener('DOMContentLoaded', function () {
    initEdTabsSection();
  });

  if (typeof Shopify !== 'undefined' && Shopify.designMode) {
    document.addEventListener('shopify:section:load', function () {
      initEdTabsSection();
    });
  }
</script>
