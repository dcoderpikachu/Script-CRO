<script>
  (function () {
    /**
     * Initializes progress ring animations within the given scope.
     * - Animates numbers and SVG rings when they enter the viewport.
     * - Prevents double initialization on dynamically loaded Shopify sections or blocks
     */
    function initProgressRing(scope = document) {
      const rings = scope.querySelectorAll('.skb-progress-ring');
      if (!rings.length) return;

      rings.forEach((ring) => {
        if (ring.dataset.initialized === 'true') return;
        ring.dataset.initialized = 'true';

        const circle = ring.querySelector('.skb-progress-ring__circle');
        const number = ring.querySelector('.skb-value');
        const target = parseInt(ring.getAttribute('data-target'), 10);

        if (!circle || !number || isNaN(target)) return;

        const radius = circle.r.baseVal.value;
        const circumference = 2 * Math.PI * radius;

        circle.style.strokeDasharray = `${circumference} ${circumference}`;
        circle.style.strokeDashoffset = circumference;

        function animate() {
          let current = 0;
          const duration = 1200;
          const stepTime = 20;
          const steps = duration / stepTime;
          const increment = target / steps;
          const offset = circumference * (1 - target / 100);

          const interval = setInterval(() => {
            current += increment;
            if (current >= target) {
              current = target;
              clearInterval(interval);
            }
            number.textContent = Math.floor(current);
            const currentOffset = circumference * (1 - current / 100);
            circle.style.strokeDashoffset = Math.max(currentOffset, offset);
          }, stepTime);
        }

        const observer = new IntersectionObserver(
          (entries) => {
            entries.forEach((entry) => {
              if (entry.isIntersecting && !entry.target.classList.contains('animated')) {
                animate();
                entry.target.classList.add('animated');
              }
            });
          },
          { threshold: 0.5 }
        );

        observer.observe(ring);
      });
    }

    initProgressRing();

    document.addEventListener('DOMContentLoaded', function () {
      initProgressRing();
    });

    if (typeof Shopify !== 'undefined' && Shopify.designMode) {
      document.addEventListener('shopify:section:load', function (event) {
        initProgressRing(event.target);
      });
      document.addEventListener('shopify:block:select', function (event) {
        initProgressRing(event.target);
      });
    }
  })();
</script>
