    function initFlickity() {
        let ed_video_autoplay =  {{ section.settings.enable_video_autoplay | default: false }};
      
        var slider = document.querySelector(".ed-flickity-slider .carousel-main");
        var sliderNav = document.querySelector(".ed-flickity-nav .carousel-nav");
        var sliderNavBottom = document.querySelector(".ed-flickity-wrapper.ed-thumbnails-bottom");

        function initNextPrevIcons() {
            // Select the next button and previous button
            const nextButton = document.querySelector('.flickity-prev-next-button.next');
            const previousButton = document.querySelector('.flickity-prev-next-button.previous');

            // SVG content for next and previous buttons
            const nextSVG = `
            <svg fill="none" xmlns="http://www.w3.org/2000/svg" width="17" height="14" viewBox="0 0 17 14">
                <path d="M0 7H15" stroke="#282828" stroke-width="2"/>
                <path d="M9 1L15 7L9 13" stroke="#282828" stroke-width="2"/>
            </svg>`;

            const previousSVG = `
            <svg fill="none" xmlns="http://www.w3.org/2000/svg" width="17" height="14" viewBox="0 0 17 14">
                <path d="M17 7H2" stroke="#282828" stroke-width="2"/>
                <path d="M8 1L2 7L8 13" stroke="#282828" stroke-width="2"/>
            </svg>`;

            // Replace the inner content of the buttons with the respective SVG
            if (nextButton) {
                nextButton.innerHTML = nextSVG;
            }

            if (previousButton) {
                previousButton.innerHTML = previousSVG;
            }
        }
        if (slider) {
            var options = {
                contain: true,
                pageDots: true,
                draggable: true,
                prevNextButtons: true,
                wrapAround: false,
                cellAlign: "left"
            };
            var flkty = new Flickity(slider, options);
            
            // Adjust height on 'select' event when the selected cell changes
            flkty.on('select', function () {
                var cells = slider.querySelectorAll('.carousel-cell');
                var carouselMain = slider;
            
                // Remove 'ed-video-active' class from carousel-main first
                carouselMain.classList.remove('ed-video-active');
            
                // Pause all videos in the carousel
                if (ed_video_autoplay) {
                    cells.forEach(cell => {
                        var video = cell.querySelector('.ed-flickity-slider-video video');
                        if (video) {
                            video.pause();
                            video.currentTime = 0; // Reset video to the beginning
                        }
                    });
                }
            
                // Check if the selected carousel cell (index) contains a video
                var selectedCell = cells[flkty.selectedIndex];
                var videoElement = selectedCell.querySelector('.ed-flickity-slider-video video');
            
                // Add 'ed-video-active' class to carousel-main and play the video if present
                if (videoElement) {
                    carouselMain.classList.add('ed-video-active');
                    if (ed_video_autoplay) {
                        videoElement.play();
                    }
                    
                }
            
                // Adjust viewport height whenever the selected cell changes
                adjustViewportHeight();
            }); 
    
            // Function to adjust the viewport height
            function adjustViewportHeight() {
                var selectedCell = slider.querySelector('.carousel-cell.is-selected');
                if (selectedCell) {
                    var thumbnail = selectedCell.querySelector('.ed-flickity-slider-item');
                    if (thumbnail) {
                        var thumbnailHeight = thumbnail.offsetHeight;
                        var viewport = slider.querySelector('.flickity-viewport');
                        var viewportNav = sliderNav.querySelector('.flickity-viewport');
                        if (viewport) {
                            viewport.style.height = thumbnailHeight + 'px';
                        }
                        if (sliderNav) {
                            sliderNav.style.width = thumbnailHeight + 'px';
                        }

                        if (sliderNavBottom) {
                            if (sliderNav) {
                                sliderNav.style.width = '100%';
                            }
                        }
                        
                    }
                }
            }

            // Initial adjustment when the DOM is loaded
            adjustViewportHeight();           


            // Disabling next-prev for video-popup
            function checkMousePosition(event) {
                var rect = slider.getBoundingClientRect();
                var mouseX = event.clientX;
                var mouseY = event.clientY;
        
                // Define the hover zone (120px x 120px from top-right)
                var topEdge = rect.top;
                var rightEdge = rect.right;
                var hoverZoneX = rightEdge - 120;
                var hoverZoneY = topEdge + 120;
        
                if (mouseX >= hoverZoneX && mouseY <= hoverZoneY) {
                    slider.classList.add("ed-video-thumbnail-active");
                } else {
                    slider.classList.remove("ed-video-thumbnail-active");
                }
            }
        
            document.addEventListener("mousemove", checkMousePosition);

            initNextPrevIcons();

    function initVideoPopup() {
        const openButtons = document.querySelectorAll(".ed-video-thumbnail");
        const videoPopup = document.querySelector(".ed-video-popup");
        const closeBtn = document.querySelector(".ed-video-popup .btn-close");
        const video = videoPopup.querySelector("video");
        video.pause();

        if (openButtons.length && videoPopup && closeBtn && video) {
            openButtons.forEach(button => {
                button.addEventListener("click", function () {
                    videoPopup.classList.add("open");
                    video.play();
                });
            });

            closeBtn.addEventListener("click", function () {
                closePopup();
            });

            videoPopup.addEventListener("click", function (event) {
                if (!video.contains(event.target) && !event.target.closest(".ed-video-thumbnail")) {
                    closePopup();
                }
            });
        }

        function closePopup() {
            videoPopup.classList.remove("open");
            video.pause();
            video.currentTime = 0; // Reset video to the beginning
        }
    }

    // Change Flickity slider based on URL variant ID
    var product = {{ product | json }};

    function initFlickityActiveItemBasedUrlVariant() {
        // Function to extract variant ID from URL
        function getVariantIdFromUrl() {
          const urlParams = new URLSearchParams(window.location.search);
          return urlParams.get('variant');
        }
      
        // Function to find the index of the slide that matches the variant image
        function findSlideIndexByImageSrc(imageSrc) {
          const slider = document.querySelector('.ed-flickity-slider .carousel-main');
          if (!slider) return -1;
      
          const cells = slider.querySelectorAll('.carousel-cell');
          for (let i = 0; i < cells.length; i++) {
            const img = cells[i].querySelector('img');
            if (img && img.src.includes(imageSrc)) {
              return i;
            }
          }
          return -1;
        }
      
        // Function to update the slider based on the current variant
        function updateSliderToVariantImage() {
          const variantId = getVariantIdFromUrl();
          if (!variantId) return;
      
          const matchedVariant = product.variants.find(variant => variant.id == variantId);
          if (!matchedVariant || !matchedVariant.featured_image) return;
      
          const variantImageSrc = matchedVariant.featured_image.src;
          const slideIndex = findSlideIndexByImageSrc(variantImageSrc);
          if (slideIndex === -1) return;
      
          const sliderElement = document.querySelector('.ed-flickity-slider .carousel-main');
          if (!sliderElement) return;
      
          const flkty = Flickity.data(sliderElement);
          if (flkty) {
            flkty.select(slideIndex);
          }
        }
      
        // Initial check on page load
        updateSliderToVariantImage();
      
        // Listen for URL changes (e.g., when variants are selected)
        window.addEventListener('popstate', updateSliderToVariantImage);
        window.addEventListener('pushstate', updateSliderToVariantImage);
        window.addEventListener('replaceState', updateSliderToVariantImage);
      
        // If your theme uses history.pushState or history.replaceState, you might need to override them to dispatch events
        const originalPushState = history.pushState;
        history.pushState = function () {
          originalPushState.apply(this, arguments);
          window.dispatchEvent(new Event('pushstate'));
          window.dispatchEvent(new Event('locationchange'));
        };
      
        const originalReplaceState = history.replaceState;
        history.replaceState = function () {
          originalReplaceState.apply(this, arguments);
          window.dispatchEvent(new Event('replacestate'));
          window.dispatchEvent(new Event('locationchange'));
        };
      
        window.addEventListener('popstate', () => {
          window.dispatchEvent(new Event('locationchange'));
        });
      
        // Listen to the custom 'locationchange' event
        window.addEventListener('locationchange', updateSliderToVariantImage);
    }
    
      
    // Run the function to load resources and initialize Swiper
    document.addEventListener("DOMContentLoaded", async function () {
        await initFlickity();
        initVideoPopup();
        initFlickityActiveItemBasedUrlVariant();
    });

    if (Shopify.designMode) {
        document.addEventListener("shopify:section:load", async function () {
            await initFlickity();
            initVideoPopup();
            initFlickityActiveItemBasedUrlVariant();
        });
    }
