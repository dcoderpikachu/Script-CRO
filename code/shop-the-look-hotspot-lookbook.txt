<script>
    function loadResource(type, url) {
        return new Promise((resolve) => {
          if (document.querySelector(`${type === "css" ? "link" : "script"}[href="${url}"], [src="${url}"]`)) {
            return resolve(); // If already loaded, resolve immediately
          }
      
          const element = document.createElement(type === "css" ? "link" : "script");
      
          if (type === "css") {
            element.rel = "stylesheet";
            element.href = url;
          } else {
            element.src = url;
            element.onload = resolve; // Resolve when script is loaded
          }
      
          document.head.appendChild(element);
          if (type === "css") resolve(); // Resolve immediately for CSS
        });
    }
      
    async function initLookBookCarousel() {
        await loadResource("css", "https://cdn.jsdelivr.net/npm/swiper@11/swiper-bundle.min.css");
        await loadResource("js", "https://cdn.jsdelivr.net/npm/swiper@11/swiper-bundle.min.js");
      
        // Initialize Swiper after script is loaded
        const swiper = new Swiper(".edLookBookCarousel", {
            cssMode: false,
            spaceBetween: 8,
            navigation: {
              nextEl: ".swiper-button-next",
              prevEl: ".swiper-button-prev",
            },
            pagination: {
              el: ".swiper-pagination",
              clickable: true
            },
            mousewheel: true,
            keyboard: true,
        });

        const buttons = document.querySelectorAll('.ed_lookbook__button');

        // Hover: Set swiper slider and update active button
        buttons.forEach((button, index) => {
            button.addEventListener('mouseenter', () => {
                swiper.slideTo(index);

                updateActiveButton(index);
            });
        });

        // Swiper: Sync button when slide changes
        swiper.on('slideChange', () => {
            updateActiveButton(swiper.activeIndex);
        });

        // Function: Update .active class
        function updateActiveButton(activeIndex) {
            buttons.forEach((btn, i) => {
                btn.classList.toggle('active', i === activeIndex);
            });
        }
    }

    document.addEventListener("DOMContentLoaded", function () {
        initLookBookCarousel();           
    });

    if (typeof Shopify !== "undefined" && Shopify.designMode) {
        document.addEventListener("shopify:section:load", function () {
            initLookBookCarousel();       
        });
    }
</script>
