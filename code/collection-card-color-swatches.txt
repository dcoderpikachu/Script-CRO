Step: 2
Go to main-collection.liquid
Add below script above the schema code:
<script>
  function initCROProductSwatches() {
    const productItems = document.querySelectorAll('product-item.product-item');

    productItems.forEach(productItem => {
      const swatchWrap = productItem.querySelector('.cro_product_swatches');
      if (!swatchWrap) return;

      const swatchButtons = swatchWrap.querySelectorAll('.cro_product_swatch__button');

      const updateProductImageFromButton = (button) => {
        const swatchInner = button.querySelector('.cro_product_swatch__inner');
        const newImageUrl = swatchInner?.dataset.background;
        if (!newImageUrl) return;

        const productImage = productItem.querySelector('.product-item__primary-image');
        if (productImage) {
          productImage.src = newImageUrl;

          const newSrcset = `
            ${newImageUrl}?width=200 200w,
            ${newImageUrl}?width=300 300w,
            ${newImageUrl}?width=400 400w,
            ${newImageUrl}?width=500 500w,
            ${newImageUrl}?width=600 600w,
            ${newImageUrl}?width=700 700w,
            ${newImageUrl}?width=800 800w,
            ${newImageUrl}?width=900 900w,
            ${newImageUrl}?width=1000 1000w,
            ${newImageUrl}?width=1100 1100w
          `.trim();

          productImage.removeAttribute('srcset');
          void productImage.offsetWidth;
          productImage.setAttribute('srcset', newSrcset);
        }
      };

      swatchButtons.forEach(button => {
        button.addEventListener('click', () => {
          swatchButtons.forEach(btn => btn.classList.remove('cro_product_swatch--selected'));
          button.classList.add('cro_product_swatch--selected');
          updateProductImageFromButton(button);
        });
      });

      const defaultSelected = swatchWrap.querySelector('.cro_product_swatch--selected');
      if (defaultSelected) {
        updateProductImageFromButton(defaultSelected);
      }
    });
  }

  document.addEventListener("DOMContentLoaded", function () {
    initCROProductSwatches();
  });

  if (typeof Shopify !== "undefined" && Shopify.designMode) {
    document.addEventListener("shopify:section:load", function () {
      initCROProductSwatches();
    });
  }

  document.addEventListener("collection:updated", function () {
    initCROProductSwatches();
  });

  // AJAX-loaded content
  const observer = new MutationObserver((mutations) => {
    for (const mutation of mutations) {
      if (mutation.addedNodes.length) {
        const productListChanged = [...mutation.addedNodes].some(node =>
          node.nodeType === 1 && node.querySelector?.('.product-item__primary-image')
        );
        if (productListChanged) {
          initCROProductSwatches();
          break;
        }
      }
    }
  });

  observer.observe(document.body, {
    childList: true,
    subtree: true
  });
</script>
