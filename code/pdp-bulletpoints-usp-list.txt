Step: 2
Create a snippet: ed-cro-bulletpoints.liquid
Paste this code:

{% if product.metafields.standard.cro_pdp_bulletpoints %}
  {% assign metafield_values = product.metafields.standard.cro_pdp_bulletpoints.value %}

  <style>
    #shopify-section-{{ section.id }} {
        {%- if block.settings.background == 'rgba(0,0,0,0)' -%}
            {%- assign section_background = settings.background -%}
        {%- else -%}
            {%- assign section_background = block.settings.background -%}
        {%- endif -%}
        {%- if block.settings.text_color == 'rgba(0,0,0,0)' -%}
            {%- assign text_color = settings.text_color -%}
        {%- else -%}
            {%- assign text_color = block.settings.text_color -%}
        {%- endif -%}
        --bulletpoints-color: {{ text_color.red }}, {{ text_color.green }}, {{ text_color.blue }};
        --bulletpoints-background: {{ section_background.red }}, {{ section_background.green }}, {{ section_background.blue }};
    }
    ul.ed_cro_bulletpoints {
        margin-top: {{ block.settings.space_top -}}px;
        margin-bottom: {{ block.settings.space_bottom -}}px;
        margin-left: 0;
        margin-right: 0;
        flex-direction: column;
        display: flex;
        gap: {{ block.settings.gap_items -}}px;
        padding: 0;
    }
    {%- if block.settings.show_border == true or block.settings.background != 'rgba(0,0,0,0)' -%}
        ul.ed_cro_bulletpoints  {
            padding: 16px;
            color: rgb(var(--bulletpoints-color));
            background-color: rgb(var(--bulletpoints-background));
            {% if block.settings.show_border %}
            border: 1px solid {{ block.settings.border_color }};
            {% endif %}
            border-radius: {{ block.settings.border_radius -}}px;
        }
    {%- endif -%}


    li.ed_cro_bulletpoints__item {
        display: flex;
        align-items: center;
        gap: {{  block.settings.icon_margin_rigght -}}px;
    }

    span.ed_cro_bulletpoints_icon {
        width: {{ block.settings.icon_width -}}px;
        height: {{ block.settings.icon_width -}}px;
        display: flex;
        align-items: center;
        justify-content: center;
        border-radius: 50px;
        {% if block.settings.icon_color == "rgba(0,0,0,0)" %}
        {% else %}
            color: {{ block.settings.icon_color }};
        {% endif %}
        {% if block.settings.icon_background == "rgba(0,0,0,0)" %}
        {% else %}
            background: {{ block.settings.icon_background }};
        {% endif %}
    }

    span.ed_cro_bulletpoints_icon svg,
    span.ed_cro_bulletpoints_icon img {
        height: auto;
        width: {{  block.settings.icon_as_size }}px;
        display: flex;
    }

    li.ed_cro_bulletpoints__item p {
        margin: 0;
        font-size: {{ block.settings.font_size -}}px;
    }
    @media (max-with: 767px) {
        li.ed_cro_bulletpoints__item p {
            font-size: {{ block.settings.font_size_mobile -}}px;
        }
    }

    ul.ed_cro_bulletpoints.ed_cro_bulletpoints--default {
        list-style: {{ block.settings.list_style }};
        {% if block.settings.background == 'rgba(0,0,0,0)' %}

        {% endif %}

        {%- endif -%}
        <p>{{ value }}</p>

    {%- endfor -%}
  </ul>
