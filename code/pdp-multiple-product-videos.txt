[Non-technical guide] Please read before making any changes
1) Duplicate your theme first (this keeps your live store safe):
    In Shopify admin go to Online Store > Themes.
    Find your current live theme, click … (Actions) > Duplicate.
    Shopify will create “Copy of <theme name>”. You can rename it (e.g., “Sandbox - Multiple Product Videos”).
    Do not publish yet; you will edit this duplicate.
2) Open the duplicated theme’s code editor:
    Next to the duplicated theme click … (Actions) > Edit code.
3) Add the section and paste the code from this file:
    In the left sidebar go to Sections > Add a new section.
    Name it: ed-multiple-product-videos.liquid
    Open that new section file, paste the code below, then Save.
4) Add the section to your Product template:
    Click Customize on the duplicated theme.
    Open a Product page in the preview.
    Click “Add section” and pick the section you created.
    Drag it where you want and Save.
5) Test safely:
    Click the buttons and confirm the video opens/plays.
    Check desktop and mobile.
6) Publish when you’re happy:
    Back in Themes, click … (Actions) on the duplicated theme > Publish.
Tips:
    If anything looks wrong, switch back to your previous live theme.
    You can always duplicate again and retry.

  <script>
    function initMPVviaMetafieldsMOdals() {
        setTimeout(() => {
            const section = document.querySelector(".multiple_product_videos_via_metafield_section");

            if (section) {
                section.querySelectorAll(".hr_mpv_via_meta_modal").forEach(modal => {
                    modal.removeAttribute("open");
                });

                section.querySelectorAll(".hr_mpv_via_meta_button").forEach(button => {
                    button.setAttribute("aria-expanded", "false");
                });
            }
        }, 000000001);
    }
    function initAutoplayModalVideoviaMetafields() {
        document.querySelectorAll("modal-content.hr_mpv_via_meta_modal").forEach(modal => {
            const video = modal.querySelector(".hr_mpv_via_meta_modal_content-video");

            if (!video) return; // Skip if there's no video inside

            const observer = new MutationObserver(mutations => {
                mutations.forEach(mutation => {
                    if (mutation.attributeName === "open") {
                        if (modal.hasAttribute("open")) {
                            video.play();
                        } else {
                            video.pause();
                            video.currentTime = 0; // Reset video to the start
                        }
                    }
                });
            });

            observer.observe(modal, { attributes: true });
        });
    }

    document.addEventListener("DOMContentLoaded", function () {
        initMPVviaMetafieldsMOdals();
        {%- if block.settings.autoplay_video -%}
            initAutoplayModalVideoviaMetafields();
        {%- endif -%}
    });

    if (typeof Shopify !== "undefined" && Shopify.designMode) {
        document.addEventListener("shopify:section:load", function () {
            initMPVviaMetafieldsMOdals();
            {%- if block.settings.autoplay_video -%}
                initAutoplayModalVideoviaMetafields();
            {%- endif -%}
        });
    }
  </script>
