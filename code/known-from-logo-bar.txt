            <script>
              (function() {
                  function loadResource(type, url) {
                      return new Promise((resolve) => {
                      if (document.querySelector(`${type === "css" ? "link" : "script"}[href="${url}"], [src="${url}"]`)) {
                          return resolve(); // Already loaded
                      }

                      const element = document.createElement(type === "css" ? "link" : "script");

                      if (type === "css") {
                          element.rel = "stylesheet";
                          element.href = url;
                      } else {
                          element.src = url;
                          element.onload = resolve;
                      }

                      document.head.appendChild(element);
                          if (type === "css") resolve(); // For CSS, resolve immediately
                      });
                  }

                  async function initLogoBar() {
                      await loadResource("css", "https://cdn.jsdelivr.net/npm/swiper@11/swiper-bundle.min.css");
                      await loadResource("js", "https://cdn.jsdelivr.net/npm/swiper@11/swiper-bundle.min.js");

                      // Wait until Swiper is available before initializing
                      if (typeof Swiper !== 'undefined') {
                          new Swiper(".ed_logo_bar_swiper_{{ section.id }}", {
                              slidesPerView: {{ section.settings.slides_per_view_mobile }},
                              spaceBetween: 15,
                              loop: true,
                              {% if section.settings.auto_play_slides %}
                              autoplay: {
                                  delay: {{ section.settings.autoplay_delay }},
                                  disableOnInteraction: false,
                                  pauseOnMouseEnter: true,
                              },
                              {% endif %}

                              navigation: {
                                  nextEl: ".ed_logo_bar_swiper-next",
                                  prevEl: ".ed_logo_bar_swiper-prev",
                              },
                              breakpoints: {
                                  768: {
                                      slidesPerView: {{ section.settings.slides_per_view }},
                                      spaceBetween: 20,
                                  }
                              },
                          });
                      }
                  }

                  document.addEventListener('DOMContentLoaded', initLogoBar);
                  if (typeof Shopify !== "undefined" && Shopify.designMode) {
                    document.addEventListener("shopify:section:load", initLogoBar);
                  }
              })();
