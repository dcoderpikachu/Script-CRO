  <script>
    // Universal Collapsible by Rana
    (function () {
      const singleOpen = true; // Set to false for multiple open accordions
      function openContent(element) {
        element.style.overflow = 'hidden';
        const height = element.scrollHeight;

        const keyframes = {
          height: ['0px', `${height}px`],
          visibility: ['hidden', 'visible'],
          opacity: element.hasAttribute('data-animate-items') ? [0, 0] : [1, 1],
        };

        const animation = element.animate(keyframes, {
          duration: 500,
          direction: 'normal',
          easing: 'cubic-bezier(0.75, 0, 0.175, 1)',
          fill: 'forwards',
        });

        animation.onfinish = () => {
          element.style.overflow = 'visible';
          element.style.height = `${height}px`;
          element.setAttribute('data-open', '');
        };

        if (element.hasAttribute('data-animate-items')) {
          element.animate(
            {
              opacity: [0, 1],
              transform: ['translateY(10px)', 'translateY(0)'],
            },
            {
              duration: 250,
              delay: 250,
              easing: 'cubic-bezier(0.75, 0, 0.175, 1)',
              fill: 'forwards',
            }
          );
        }
      }

      function closeContent(element) {
        const height = element.scrollHeight;
        element.style.overflow = 'hidden';

        const animation = element.animate(
          {
            height: [`${height}px`, '0px'],
            visibility: ['visible', 'hidden'],
            opacity: element.hasAttribute('data-animate-items') ? [0, 0] : [1, 0],
          },
          {
            duration: 500,
            direction: 'normal',
            easing: 'cubic-bezier(0.75, 0, 0.175, 1)',
            fill: 'forwards',
          }
        );

        animation.onfinish = () => {
          element.style.overflow = 'hidden';
          element.removeAttribute('data-open');
        };
      }

      function initTechnicalSpecCollapsible(scope = document) {
        const wraps = scope.querySelectorAll('ed-collapsible-wrap');

        wraps.forEach((wrap) => {
          const toggles = wrap.querySelectorAll('.ed-collapsible-toggle');

          toggles.forEach((toggle) => {
            if (toggle.dataset.initialized === 'true') return;

            toggle.dataset.initialized = 'true';

            const contentId = toggle.getAttribute('aria-controls');
            const content = wrap.querySelector(`#${contentId}`);
            if (!content) return;

            // Keep initial state as-is (open or closed)
            const isInitiallyOpen = content.hasAttribute('data-open');
            toggle.setAttribute('aria-expanded', String(isInitiallyOpen));
            toggle.classList.toggle('is-open', isInitiallyOpen);

            toggle.addEventListener('click', () => {
              const isOpen = toggle.getAttribute('aria-expanded') === 'true';

              // If singleOpen is true, close all other open accordions in the same wrap
              if (singleOpen && !isOpen) {
                const openToggles = wrap.querySelectorAll('.ed-collapsible-toggle[aria-expanded="true"]');
                openToggles.forEach((otherToggle) => {
                  if (otherToggle !== toggle) {
                    const otherContentId = otherToggle.getAttribute('aria-controls');
                    const otherContent = wrap.querySelector(`#${otherContentId}`);
                    if (otherContent && otherContent.hasAttribute('data-open')) {
                      otherToggle.setAttribute('aria-expanded', 'false');
                      otherToggle.classList.remove('is-open');
                      closeContent(otherContent);
                    }
                  }
                });
              }

              toggle.setAttribute('aria-expanded', String(!isOpen));
