  <script>
    // Play/Pause video in collapsible box
    (function () {
      const container = document.querySelector('.ed_cro_pdp_collapsible_box__video');
      if (!container) return; // Exit if container is not found

      const playButton = container.querySelector('.cro_pdp_collapsible_box_video_control_btn');
      const video = container.querySelector('video');

      // Exit if button or video is missing
      if (!playButton || !video) return;

      // Play video and show active state
      playButton.addEventListener('click', function (e) {
        e.stopPropagation();
        video.play();
        container.classList.add('active');
      });

      // Pause video and remove active state
      video.addEventListener('click', function () {
        if (!video.paused) {
          video.pause();
          container.classList.remove('active');
        }
      });
    })();
  </script>
