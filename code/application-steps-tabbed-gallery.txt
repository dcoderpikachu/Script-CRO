<script>
    function initApplicationTabs(container) {
        const buttons = container.querySelectorAll(".ed_application_step_img");
        const steps = container.querySelectorAll(".ed_application_step");
        const progress = container.querySelector("progress");
      
        if (!buttons.length || !steps.length || !progress) return;
      
        buttons.forEach((btn, index) => {
          btn.addEventListener("click", () => {
            buttons.forEach(b => b.classList.remove("active"));
            steps.forEach(step => step.style.display = "none");
      
            btn.classList.add("active");
            steps[index].style.display = "block";
            progress.value = ((index + 1) / buttons.length) * 100;
          });
        });
    }
      
    document.addEventListener("DOMContentLoaded", function () {
        document.querySelectorAll(".ed_application_steps_wrapper").forEach(initApplicationTabs);
    });
    
    if (Shopify.designMode) {
        document.addEventListener('shopify:section:load', function (e) {
            const container = e.target.querySelector('.ed_application_steps_wrapper');
            if (container) initApplicationTabs(container);
        });
    } 
</script>
