Add below code at section schema
Paste this code:
{
      "type": "header",
      "content": "Buy Box"
},
{
    "type": "checkbox",
    "id": "enable_buy_box",
    "label": "Enable Buy Box",
    "default": false
},
{
    "type": "checkbox",
    "id": "sticky_buy_box",
    "label": "Sticky Buy Box",

},
{
    "type": "color",
    "id": "buy_box_border_color",

},
{
    "type": "color",
    "id": "buy_box_background_color",

Step: 2
Go to snippet: product-info.liquid 
Search for  product__info class and add below code
Paste this code:

{%- if section.settings.enable_buy_box -%}
  <style>
    #shopify-section-{{ section.id }} {
      {%- if section.settings.buy_box_border_color == 'rgba(0,0,0,0)' %}

        {%- else -%}
          {%- assign border_color = section.settings.buy_box_border_color -%}
      {%- endif -%}
  
      {%- if section.settings.buy_box_background_color == 'rgba(0,0,0,0)' -%}
        {%- assign background_color = settings.background -%}
        {%- else -%}
          {%- assign background_color = section.settings.buy_box_background_color -%}
      {%- endif -%}
      --buybox-border-color: {{ border_color.red }}, {{ border_color.green }}, {{ border_color.blue }};
      --buybox-background-color: {{ background_color.red }}, {{ background_color.green }}, {{ background_color.blue }};
    }
    {%- if section.settings.sticky_buy_box -%}
      .product__info {
        position: sticky;
        top: 100px;
      }
    {%- endif -%}

  
    .ed-pdp-buy-box {
      border: 1px solid rgb(var(--buybox-border-color));

          margin-top: 0;
      }
    }
  </style>  
{%- endif -%}
