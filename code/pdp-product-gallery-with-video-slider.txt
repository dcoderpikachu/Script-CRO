<script>
  (function () {
    // Init Swiper slider
    function initImageWithVideoSwiper() {
      // Initialize Swiper after script is loaded
      var swiper = new Swiper('.edImageWithVideoSwiper', {
        slidesPerView: 1,
        slidesPerGroup: 1,
        spaceBetween: 2,
        loop: false,
        speed: 600,
        navigation: {
          nextEl: '.swiper-button-next',
          prevEl: '.swiper-button-prev',
        },
        pagination: {
          el: '.swiper-pagination',
          clickable: true,
        },
        scrollbar: {
          el: '.swiper-scrollbar',
          hide: true,
        },
        breakpoints: {
          1024: {
            slidesPerView: 2,
            slidesPerGroup: 2,
            spaceBetween: 2,
            autoHeight: true,
          },
        },
      });
    }

    // Initialize custom Video Controls
    function initCustomVideoControls(wrapperSelector = '.edImageWithVideoSwiper__video') {
      const videoWrappers = document.querySelectorAll(wrapperSelector);
      if (!videoWrappers.length) return;

      videoWrappers.forEach((wrapper) => {
        const video = wrapper.querySelector('video');
        const playBtn = wrapper.querySelector('.edImageWithVideoSwiper__video_play');
        const pauseBtn = wrapper.querySelector('.edImageWithVideoSwiper__video_pause');
        const fullscreenBtn = wrapper.querySelector('.edImageWithVideoSwiper__video_fullscreen');

        if (!video) return;

        // Pause & Play
        if (playBtn && pauseBtn) {
          playBtn.addEventListener('click', () => {
            video.play();
            playBtn.classList.remove('active');
            pauseBtn.classList.add('active');
          });

          pauseBtn.addEventListener('click', () => {
            video.pause();
            pauseBtn.classList.remove('active');
            playBtn.classList.add('active');
          });

          video.addEventListener('pause', () => {
            pauseBtn.classList.remove('active');
            playBtn.classList.add('active');
          });

          video.addEventListener('play', () => {
            playBtn.classList.remove('active');
            pauseBtn.classList.add('active');
          });
        }

        // Fullscreen Handling
        if (fullscreenBtn) {
          fullscreenBtn.addEventListener('click', () => {
            const elem = wrapper;

            if (document.fullscreenElement || document.webkitFullscreenElement || document.msFullscreenElement) {
              // Exit fullscreen
              if (document.exitFullscreen) {
                document.exitFullscreen();
              } else if (document.webkitExitFullscreen) {
                document.webkitExitFullscreen();
              } else if (document.msExitFullscreen) {
                document.msExitFullscreen();
              }
            } else {
              // Enter fullscreen
              if (elem.requestFullscreen) {
                elem.requestFullscreen();
              } else if (elem.webkitRequestFullscreen) {
                elem.webkitRequestFullscreen();
              } else if (elem.msRequestFullscreen) {
                elem.msRequestFullscreen();
              } else {
                console.warn('Fullscreen API not supported');
              }
            }
          });

          // Track Fullscreen Changes
          document.addEventListener('fullscreenchange', () => {
            if (document.fullscreenElement === wrapper) {
              wrapper.classList.add('fullscreen_view');
            } else {
              wrapper.classList.remove('fullscreen_view');
            }
          });

          // Support for Safari and legacy prefixes
          document.addEventListener('webkitfullscreenchange', () => {
            if (document.webkitFullscreenElement === wrapper) {
              wrapper.classList.add('fullscreen_view');
            } else {
              wrapper.classList.remove('fullscreen_view');
            }
          });

          document.addEventListener('msfullscreenchange', () => {
            if (document.msFullscreenElement === wrapper) {
              wrapper.classList.add('fullscreen_view');
            } else {
              wrapper.classList.remove('fullscreen_view');
            }
          });
        }
      });
    }

    // Initialize Video Modal popup
    function initVideoPopup() {
      const openButtons = document.querySelectorAll('.ed-video-thumbnail');
      const videoPopup = document.querySelector('.ed-video-popup');
      const closeBtn = document.querySelector('.ed-video-popup .btn-close');
      const video = videoPopup.querySelector('video');
      video.pause();

      if (openButtons.length && videoPopup && closeBtn && video) {
        openButtons.forEach((button) => {
          button.addEventListener('click', function () {
            videoPopup.classList.add('open');
            video.play();
          });
        });

        closeBtn.addEventListener('click', function () {
          closePopup();
        });

        videoPopup.addEventListener('click', function (event) {
          if (!video.contains(event.target) && !event.target.closest('.ed-video-thumbnail')) {
            closePopup();
          }
        });
      }

      function closePopup() {
        videoPopup.classList.remove('open');
        video.pause();
        video.currentTime = 0; // Reset video to the beginning
      }
    }

    // Run the function to load resources and initialize Swiper
    document.addEventListener('DOMContentLoaded', async function () {
      await initImageWithVideoSwiper();
      initCustomVideoControls();
      initVideoPopup();
    });

    if (Shopify.designMode) {
      document.addEventListener('shopify:section:load', async function () {
        await initImageWithVideoSwiper();
        initCustomVideoControls();
        initVideoPopup();
      });
    }
  })();
</script>

<script>
  /* Changing slide based on URL variant ID */
  (function () {
    // Pass in the product JSON as a parameter
    function setupVariantSwiperSync(product, swiperSelector = '.edImageWithVideoSwiper') {
      function getVariantIdFromUrl() {
        const params = new URLSearchParams(window.location.search);
        return params.get('variant');
      }

      function normalizeSrc(src) {
        return src?.split('?')[0].replace(/^https?:/, '');
      }

      function findSlideIndex(sliderEl, variantImgSrc) {
        const slides = sliderEl.querySelectorAll('.swiper-slide img');
        const normalizedTarget = normalizeSrc(variantImgSrc);

        for (let i = 0; i < slides.length; i++) {
          const img = slides[i];
          if (normalizeSrc(img.src).includes(normalizedTarget)) {
            return i;
          }
        }
        return -1;
      }

      function updateSliderForVariant(sliderEl) {
        const swiper = sliderEl.swiper;
        if (!swiper) return;

        const variantId = getVariantIdFromUrl();
        if (!variantId) return;

        const variant = product.variants.find((v) => v.id == variantId);
        if (!variant || !variant.featured_image) return;

        const index = findSlideIndex(sliderEl, variant.featured_image.src);
        if (index !== -1) swiper.slideTo(index);
      }

      function attachLocationChangeEvents() {
        if (window._variantSwiperEventsAttached) return;
        window._variantSwiperEventsAttached = true;

        const origPush = history.pushState;
        history.pushState = function () {
          origPush.apply(this, arguments);
          window.dispatchEvent(new Event('locationchange'));
        };

        const origReplace = history.replaceState;
        history.replaceState = function () {
          origReplace.apply(this, arguments);
          window.dispatchEvent(new Event('locationchange'));
        };

        window.addEventListener('popstate', () => {
          window.dispatchEvent(new Event('locationchange'));
        });
      }

      function init() {
        const sliders = document.querySelectorAll(swiperSelector);
        if (!sliders.length) return;

        sliders.forEach((sliderEl) => {
          updateSliderForVariant(sliderEl);
        });

        window.addEventListener('locationchange', () => {
          sliders.forEach((sliderEl) => {
            updateSliderForVariant(sliderEl);
          });
        });
      }

      attachLocationChangeEvents();
      init();

      // Shopify editor live reload
      if (Shopify.designMode) {
        document.addEventListener('shopify:section:load', init);
      }
    }

    // Run the function to load resources and initialize Swiper
    document.addEventListener('DOMContentLoaded', async function () {
      const product = {{ product | json }};
      setupVariantSwiperSync(product);
    });

    if (Shopify.designMode) {
      document.addEventListener('shopify:section:load', async function () {
        const product = {{ product | json }};
      setupVariantSwiperSync(product);
      });
    }
  })();
</script>

Paste the JS BEFORE the closing </body> tag (on its own line):
<script src="https://cdn.jsdelivr.net/npm/swiper@11/swiper-bundle.min.js"></script>

Notes:
- If your theme already loads Swiper, skip adding it again to avoid duplicates.
- If your store blocks external CDNs, upload the Swiper files to Assets and include them:
  CSS: {{ 'swiper-bundle.min.css' | asset_url | stylesheet_tag }}
  JS:  {{ 'swiper-bundle.min.js'  | asset_url | script_tag }}
  (Shopify admin > Online Store > Themes > Edit code > Assets > Add files)



Step: 3
Insert the gallery into your product page (by theme)

Where to edit (common theme file names/paths):
- Shopify free OS 2.0 themes (Dawn, Refresh, Sense, Craft, Origin, Taste, Studio, Publisher): sections/main-product.liquid
- Many popular premium themes (Impulse, Motion, Prestige, Warehouse, Streamline, etc.): sections/product-template.liquid or sections/product.liquid
- If you can't find the file above: open templates/product.json and note the main section type (often "main-product"), then open the matching file in sections/ (e.g., sections/main-product.liquid)

How to locate the exact insertion point:
1) Open the product section file from the list above.
2) Find and remove or comment out the theme’s default media gallery block. Look for one of these patterns:
   - {%- render 'product-media-gallery' -%} or {% section 'product-media-gallery' %}
   - {% section 'product-gallery' %} or a similarly named gallery section/snippet
   - <media-gallery ...> ... </media-gallery>
   - A loop like: {% for media in product.media %} ... {% endfor %}
   - A container with classes like product__media or media-gallery
3) Insert our render line in the exact place where the old gallery was (inside the left media column/container, not inside the product form/right column):
   {%- render 'ed-product-image-with-video', product: product, section: section -%}

Notes:
- Keeping the render inside the same wrapper ensures layout and CSS sizing remain correct.
- If two galleries appear, you missed a leftover gallery include—remove/comment it out so only our render remains.

Troubleshooting / FAQ
- I see two galleries: Hide/remove the theme’s default Product media block or comment out its markup, leaving only the custom render line.
- "Swiper is not defined": Ensure the JS tag is added BEFORE </body> in layout/theme.liquid and that the CDN isn’t blocked. If blocked, upload Swiper files to Assets and include via asset_url (see Step 2 notes).
- Video doesn’t start: Make sure you uploaded a video in the product metafield and that the video file is processed. Autoplay can be restricted by browsers; use the play button if needed.
- Nothing shows: Verify the snippet name is exactly ed-product-image-with-video.liquid and that you included the render line in the correct template/section.
