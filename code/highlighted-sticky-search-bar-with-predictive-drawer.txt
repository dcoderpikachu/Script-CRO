<script>
  
document.addEventListener('DOMContentLoaded', function () {
const currencyFormat = window.themeVariables.settings.currencyCodeEnabled
          ? window.themeVariables.settings.moneyWithCurrencyFormat
          : window.themeVariables.settings.moneyFormat;
function formatMoney(cents, format = "") {
    if (typeof cents === "string") {
      cents = cents.replace(".", "");
    }
    const placeholderRegex = /\{\{\s*(\w+)\s*\}\}/,
      formatString = format || window.themeVariables.settings.moneyFormat;
    function defaultTo(value2, defaultValue) {
      return value2 == null || value2 !== value2 ? defaultValue : value2;
    }
    function formatWithDelimiters(number, precision, thousands, decimal) {
      precision = defaultTo(precision, 2);
      thousands = defaultTo(thousands, ",");
      decimal = defaultTo(decimal, ".");
      if (isNaN(number) || number == null) {
        return 0;
      }
      number = (number / 100).toFixed(precision);
      let parts = number.split("."),
        dollarsAmount = parts[0].replace(
          /(\d)(?=(\d\d\d)+(?!\d))/g,
          "$1" + thousands
        ),
        centsAmount = parts[1] ? decimal + parts[1] : "";
      return dollarsAmount + centsAmount;
    }
    let value = "";
    switch (formatString.match(placeholderRegex)[1]) {
      case "amount":
        value = formatWithDelimiters(cents, 2);
        break;
      case "amount_no_decimals":
        value = formatWithDelimiters(cents, 0);
        break;
      case "amount_with_space_separator":
        value = formatWithDelimiters(cents, 2, " ", ".");
        break;
      case "amount_with_comma_separator":
        value = formatWithDelimiters(cents, 2, ".", ",");
        break;
      case "amount_with_apostrophe_separator":
        value = formatWithDelimiters(cents, 2, "'", ".");
        break;
      case "amount_no_decimals_with_comma_separator":
        value = formatWithDelimiters(cents, 0, ".", ",");
        break;
      case "amount_no_decimals_with_space_separator":
        value = formatWithDelimiters(cents, 0, " ");
        break;
      case "amount_no_decimals_with_apostrophe_separator":
        value = formatWithDelimiters(cents, 0, "'");
        break;
    }
    if (formatString.indexOf("with_comma_separator") !== -1) {
      return formatString.replace(placeholderRegex, value);
    } else {
      return formatString.replace(placeholderRegex, value);
    }
  }
  
const STRINGS = {
  typeToSearch: 'Tippen Sie, um zu suchen',
  searching: 'Wird gesucht...',
  noProducts: 'Keine Produkte gefunden',
  noCollections: 'Keine Kollektionen gefunden',
  noPages: 'Keine Seiten gefunden',
  noArticles: 'Keine Artikel gefunden',
  noResults: 'Keine Ergebnisse gefunden',
  errorLoading: 'Fehler beim Laden der Ergebnisse',
  viewAll: 'Alle Ergebnisse anzeigen'
};

  const CONFIG = {
    debounceTime: 300,
    productImageSize: '300x300',
    articleImageSize: '300x200',
    excerptLength: 100,
    resultsLimit: 5
  };

  const searchDrawer = document.getElementById('psy-search-drawer');
  if (!searchDrawer) return;

  const searchInput = searchDrawer.querySelector('#search-input');
  const searchSubmit = searchDrawer.querySelector('.search-drawer__submit');
  const viewAllResults = searchDrawer.querySelector('#view-all-results');
  const closeButton = searchDrawer.querySelector('.search-drawer__close');
  const overlay = searchDrawer.querySelector('.search-drawer__overlay');
  const tabsContainer = searchDrawer.querySelector('.search-drawer__tabs');
  const tabContents = searchDrawer.querySelectorAll('.search-drawer__tab-content');
  const tabs = searchDrawer.querySelectorAll('.search-drawer__tab');

  let currentRequest = null;
  let debounceTimer = null;

  function openSearchDrawer() {
    searchDrawer.classList.add('active');
    searchInput.focus();
    tabsContainer.style.display = 'none';
  }

  function closeSearchDrawer() {
    searchDrawer.classList.remove('active');
    searchInput.value = '';
    clearResults();
    tabsContainer.style.display = 'none';
  }

  function clearResults() {
    searchDrawer.querySelector('.search-drawer__products').innerHTML = '';
    searchDrawer.querySelector('.search-drawer__collections').innerHTML = '';
    searchDrawer.querySelector('.search-drawer__pages').innerHTML = '';
    searchDrawer.querySelector('.search-drawer__articles').innerHTML = '';
    viewAllResults.disabled = true;
  }

  function setupTabEvents() {
    tabs.forEach(tab => {
      tab.addEventListener('click', function () {
        const tabId = this.getAttribute('data-tab');

        tabs.forEach(t => t.classList.remove('active'));
        this.classList.add('active');

        tabContents.forEach(content => content.classList.remove('active'));
        const tabContent = searchDrawer.querySelector(`#${tabId}-results`);
        if (tabContent) tabContent.classList.add('active');
      });
    });
  }

  closeButton.addEventListener('click', closeSearchDrawer);
  overlay.addEventListener('click', closeSearchDrawer);

  viewAllResults.addEventListener('click', function () {
    if (searchInput.value.trim()) {
      window.location.href = `/search?q=${encodeURIComponent(searchInput.value.trim())}`;
    }
  });

  searchInput.addEventListener('keyup', function (e) {
    if (e.key === 'Enter' && searchInput.value.trim()) {
      window.location.href = `/search?q=${encodeURIComponent(searchInput.value.trim())}`;
    }
  });

  searchSubmit.addEventListener('click', function () {
    if (searchInput.value.trim()) {
      window.location.href = `/search?q=${encodeURIComponent(searchInput.value.trim())}`;
    }
  });

  function renderProduct(product) {
    return `
      <div class="search-drawer__product">
        <a href="${product.url}" class="search-drawer__product-link">
          ${product.image ? `
            <img
              src="${product.image}"
              alt="${product.title}"
              class="search-drawer__product-image"
              loading="lazy"
              width="${CONFIG.productImageSize.split('x')[0]}"
              height="${CONFIG.productImageSize.split('x')[1]}"
            >
          ` : ''}
          <h4 class="search-drawer__product-title">${product.title}</h4>
          <div class="search-drawer__product-price">
            ${formatMoney(product.price, currencyFormat)}
          </div>
        </a>
      </div>
    `;
  }

  function renderCollection(collection) {
    return `
      <div class="search-drawer__collection">
        <a href="${collection.url}">
          <h4 class="search-drawer__collection-title">${collection.title}</h4>
        </a>
      </div>
    `;
  }

  function renderPage(page) {
    return `
      <div class="search-drawer__page">
        <a href="${page.url}">
          <h4 class="search-drawer__page-title">${page.title}</h4>
        </a>
      </div>
    `;
  }

  function renderArticle(article) {
    return `
      <div class="search-drawer__article">
        <a href="${article.url}">
          ${article.image ? `
            <img
              src="${article.image}"
              alt="${article.title}"
              class="search-drawer__article-image"
              loading="lazy"
              width="${CONFIG.articleImageSize.split('x')[0]}"
              height="${CONFIG.articleImageSize.split('x')[1]}"
            >
          ` : ''}
          <h4 class="search-drawer__article-title">${article.title}</h4>
          <p class="search-drawer__article-excerpt">
            ${article.excerpt || ''}
          </p>
        </a>
      </div>
    `;
  }

  function performSearch(query) {
    if (!query || query.length < 1) {
      clearResults();
      searchDrawer.querySelectorAll('.search-drawer__tab-content > div').forEach(content => {
        content.innerHTML = `<div class="search-drawer__empty">${STRINGS.typeToSearch}</div>`;
      });
      tabsContainer.style.display = 'none';
      return;
    }

    if (currentRequest) {
      currentRequest.abort();
    }

    viewAllResults.disabled = true;
    tabsContainer.style.display = 'none';

    searchDrawer.querySelectorAll('.search-drawer__tab-content > div').forEach(container => {
      container.innerHTML = `<div class="search-drawer__empty">${STRINGS.searching}</div>`;
    });

    const endpoint = `/search/suggest.json?q=${encodeURIComponent(query)}&resources[type]=product,collection,page,article&resources[limit]=${CONFIG.resultsLimit}&resources[options][unavailable_products]=hide`;

    currentRequest = fetch(endpoint, {
      headers: {
        'Content-Type': 'application/json',
        'Accept': 'application/json'
      }
    })
      .then(response => {
        if (!response.ok) throw new Error('Network response was not ok');
        return response.json();
      })
      .then(data => {
        currentRequest = null;
        let hasResults = false;
        let activeTabSet = false;

        const productsContainer = searchDrawer.querySelector('.search-drawer__products');
        if (data.resources.results.products.length > 0) {
          productsContainer.innerHTML = '';
          data.resources.results.products.forEach(product => {
            const formattedProduct = {
              ...product,
              price: product.price,
              compare_at_price: product.compare_at_price || null,
              image: product.image ? product.image.replace(/\{.*?\}/, CONFIG.productImageSize) : null
            };
            productsContainer.innerHTML += renderProduct(formattedProduct);
          });
          searchDrawer.querySelector('.search-drawer__tab[data-tab="products"]').style.display = 'block';
          hasResults = true;

          if (!activeTabSet) {
            searchDrawer.querySelector('.search-drawer__tab[data-tab="products"]').classList.add('active');
            const tabContent = searchDrawer.querySelector('#products-results');
            if (tabContent) tabContent.classList.add('active');
            activeTabSet = true;
          }
        } else {
          productsContainer.innerHTML = `<div class="search-drawer__empty" style="display:none">${STRINGS.noProducts}</div>`;
          searchDrawer.querySelector('.search-drawer__tab[data-tab="products"]').style.display = 'none';
        }

        const collectionsContainer = searchDrawer.querySelector('.search-drawer__collections');
        if (data.resources.results.collections.length > 0) {
          collectionsContainer.innerHTML = '';
          data.resources.results.collections.forEach(collection => {
            collectionsContainer.innerHTML += renderCollection(collection);
          });
          searchDrawer.querySelector('.search-drawer__tab[data-tab="collections"]').style.display = 'block';
          hasResults = true;
        } else {
          collectionsContainer.innerHTML = `<div class="search-drawer__empty" style="display:none">${STRINGS.noCollections}</div>`;
          searchDrawer.querySelector('.search-drawer__tab[data-tab="collections"]').style.display = 'none';
        }

        const pagesContainer = searchDrawer.querySelector('.search-drawer__pages');
        if (data.resources.results.pages.length > 0) {
          pagesContainer.innerHTML = '';
          data.resources.results.pages.forEach(page => {
            pagesContainer.innerHTML += renderPage(page);
          });
          searchDrawer.querySelector('.search-drawer__tab[data-tab="pages"]').style.display = 'block';
          hasResults = true;
        } else {
          pagesContainer.innerHTML = `<div class="search-drawer__empty" style="display:none">${STRINGS.noPages}</div>`;
          searchDrawer.querySelector('.search-drawer__tab[data-tab="pages"]').style.display = 'none';
        }

        const articlesContainer = searchDrawer.querySelector('.search-drawer__articles');
        if (data.resources.results.articles.length > 0) {
          articlesContainer.innerHTML = '';
          data.resources.results.articles.forEach(article => {
            const formattedArticle = {
              ...article,
              image: article.image ? article.image.replace(/\{.*?\}/, CONFIG.articleImageSize) : null,
              excerpt: article.body ? article.body.substring(0, CONFIG.excerptLength) + '...' : ''
            };
            articlesContainer.innerHTML += renderArticle(formattedArticle);
          });
          searchDrawer.querySelector('.search-drawer__tab[data-tab="articles"]').style.display = 'block';
          hasResults = true;
        } else {
          articlesContainer.innerHTML = `<div class="search-drawer__empty" style="display:none">${STRINGS.noArticles}</div>`;
          searchDrawer.querySelector('.search-drawer__tab[data-tab="articles"]').style.display = 'none';
        }

        if (hasResults) {
          tabsContainer.style.display = 'flex';
          viewAllResults.disabled = false;
          viewAllResults.textContent = STRINGS.viewAll;

          if (!activeTabSet) {
            const firstVisibleTab = searchDrawer.querySelector('.search-drawer__tab[style="display: block;"]');
            if (firstVisibleTab) {
              const tabId = firstVisibleTab.getAttribute('data-tab');
              firstVisibleTab.classList.add('active');
              const tabContent = searchDrawer.querySelector(`#${tabId}-results`);
              if (tabContent) tabContent.classList.add('active');
              activeTabSet = true;
            }
          }
        } else {
          tabsContainer.style.display = 'none';
          viewAllResults.disabled = true;
          searchDrawer.querySelector('.search-drawer__products').innerHTML = `<div class="search-drawer__empty">${STRINGS.noResults}</div>`;
        }
      })
      .catch(error => {
        if (error.name !== 'AbortError') {
          console.error('Search error:', error);
          searchDrawer.querySelectorAll('.search-drawer__tab-content > div').forEach(container => {
            container.innerHTML = `<div class="search-drawer__empty">${STRINGS.errorLoading}</div>`;
          });
          tabsContainer.style.display = 'none';
          viewAllResults.disabled = true;
        }
      });
  }

  function debounceSearch() {
    clearTimeout(debounceTimer);
    debounceTimer = setTimeout(() => {
      performSearch(searchInput.value.trim());
    }, CONFIG.debounceTime);
  }

  searchInput.addEventListener('input', debounceSearch);

  setupTabEvents();
  window.openSearchDrawer = openSearchDrawer;

  document.querySelectorAll('.psy-search-drawer-toggle').forEach(function(toggle) {
  toggle.addEventListener('click', function(event) {
    event.preventDefault(); 
    if (searchDrawer.classList.contains('active')) {
      closeSearchDrawer();
    } else {
      openSearchDrawer();
    }
  });
});

    

});
