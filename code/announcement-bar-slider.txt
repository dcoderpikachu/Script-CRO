    <script>
      function loadResource(type, url) {
        return new Promise((resolve) => {
          if (document.querySelector(`${type === 'css' ? 'link' : 'script'}[href="${url}"], [src="${url}"]`)) {
            return resolve(); // If already loaded, resolve immediately
          }

          const element = document.createElement(type === 'css' ? 'link' : 'script');

          if (type === 'css') {
            element.rel = 'stylesheet';
            element.href = url;
          } else {
            element.src = url;
            element.onload = resolve; // Resolve when script is loaded
          }

          document.head.appendChild(element);
          if (type === 'css') resolve(); // Resolve immediately for CSS
        });
      }

      async function initAnnouncementSwiper() {
        await loadResource('css', 'https://cdn.jsdelivr.net/npm/swiper@11/swiper-bundle.min.css');
        await loadResource('js', 'https://cdn.jsdelivr.net/npm/swiper@11/swiper-bundle.min.js');

        // Initialize Swiper after script is loaded
        new Swiper('.edAnnouncementBar', {
          slidesPerView: 1,
          spaceBetween: 10,
          loop: {{ section.settings.autoplay }},
          navigation: {
            nextEl: '.ed_announcement_bar_swiper-next',
            prevEl: '.ed_announcement_bar_swiper-prev',
          },
          breakpoints: {
            768: {
              slidesPerView: {{ section.settings.slides_per_view_desktop }},
              spaceBetween: 10,
            },
          },
        });
      }
      initAnnouncementSwiper();
      if (typeof Shopify !== 'undefined' && Shopify.designMode) {
        document.addEventListener('shopify:section:load', function () {
          initAnnouncementSwiper();
        });
      }
    </script>
